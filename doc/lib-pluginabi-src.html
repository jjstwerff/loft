src/pluginabi.loft

// pluginabi — the §9c.2 plugin call protocol (Track L4, step 1).
//
// WHAT THIS IS. A plugin implements six pure functions (§9c.2). The host does NOT call them as
// six named wasm exports — that would need a codegen feature loft does not have, and inventing a
// per-plugin `(ptr,len)` ABI beside loft's own was the mistake the first two design drafts made.
// Instead the host drives the plugin over the byte channel loft ALREADY exposes on every wasm it
// compiles (`loft_io.loft_host_input_len` / `loft_host_input_copy` in, `loft_host_print` out,
// pumped by the host) — exactly the shape `tools/wasm_debug_client.mjs` drives the debugger with.
//
// So the six operations are FRAMES on one channel, and this file is that frame vocabulary. It is
// the only thing either side needs to agree on: no ABI, no shim, no runtime. See
// doc/PLUGIN_RUNTIME.md § "P3 results".
//
// WHY CBOR AND NOT A TEXT PROTOCOL. The debugger's `D!:` frames are text because breakpoints and
// expressions are text. Plugin payloads are opaque BYTES (state, ops, snapshots — §9c.2 says the
// plugin owns their meaning), and they must survive a round-trip unchanged; canonical CBOR
// already carries byte strings losslessly here and is what every other record in this project
// uses. Reusing it means the §8.15 determinism story needs no second encoding to reason about.
//
// Run: `loft test`.
use cbor;     // the published canonical-CBOR codec
use crypto;   // base64 <-> bytes

// ---------- CBOR helpers ----------
// Inlined rather than depending on a project-local CBOR wrapper: a published protocol library
// must stand on published deps alone.
//
// ALL PREFIXED `pa_`, and that is not cosmetic. A library's PRIVATE helpers still share a name
// space with every other library in a consumer's program, so a plain `decode_value` here collides
// with a consumer's own `decode_value` — which is exactly what happened: the collision was
// invisible in this package's own tests and in a minimal consumer, and only surfaced in a program
// that used both. A published library must not impose names a consumer cannot see coming.
fn pa_ctext(s: text) -> CborValue        { return CText { value: s }; }
fn pa_cbool(b: boolean) -> CborValue     { return CBool { value: b }; }
fn pa_cbytes(b64: text) -> CborValue { return CBytes { value: base64_to_bytes(b64) }; }
fn pa_te(k: text, v: CborValue) -> CborEntry { return CborEntry { key: CText { value: k }, value: v }; }

fn pa_decode(bytes: vector<u8>) -> CborValue {
  d = decode(bytes);
  v = d.value;      // local copy — returning d.value directly corrupts native (loft #425)
  return v;
}
fn pa_decode_ok(bytes: vector<u8>) -> boolean { d = decode(bytes); return d.ok; }

fn pa_get(m: CborValue, key: text) -> CborValue {
  // The `match` is the function's final EXPRESSION (fallback `CNull` in the arm tails), NOT a
  // statement followed by `return CNull`. Those two forms are logically identical, but the
  // statement form corrupts the early-returned `r` once its embedded vector is large enough
  // (~54 bytes) — on interpret AND native — because `r` is a heap CborValue and the fall-through
  // path frees it while it is still the live return value. See loft#651. The expression form is
  // the workaround until that is fixed; every plugin frame read funnels through here.
  match m {
    CMap { entries } => {
      for i in 0..len(entries) {
        e = entries[i] ?? CborEntry { key: CNull, value: CNull };
        match e.key { CText { value } => { if value == key { r = e.value; return r; } }, _ => {} }
      }
      CNull
    },
    _ => CNull,
  }
}
fn pa_text(m: CborValue, key: text) -> text {
  match pa_get(m, key) { CText { value } => { return "{value}"; }, _ => {} }
  return "";
}
fn pa_bytes(m: CborValue, key: text) -> text {
  match pa_get(m, key) { CBytes { value } => { return bytes_to_base64(value); }, _ => {} }
  return "";
}

// ---------- the operation vocabulary (§9c.2) ----------
// The six, and only these six. `valid_op` is the gate: a host that receives an unknown op must
// reject rather than guess, because a plugin and host disagreeing about the vocabulary is a
// version skew, not a request to be interpreted.
pub const OP_INITIAL_STATE: text = "initial_state";
pub const OP_APPLY_OP:      text = "apply_op";
pub const OP_MAKE_OP:       text = "make_op";
pub const OP_RENDER:        text = "render";
pub const OP_SNAPSHOT:      text = "snapshot";
pub const OP_LOAD_SNAPSHOT: text = "load_snapshot";

pub fn valid_op(op: text) -> boolean {
  return (op == OP_INITIAL_STATE) || (op == OP_APPLY_OP) || (op == OP_MAKE_OP)
      || (op == OP_RENDER) || (op == OP_SNAPSHOT) || (op == OP_LOAD_SNAPSHOT);
}

// Which operations read the `arg` slot. Recorded here rather than in each host so the two sides
// cannot drift: a host that sends `arg` for `snapshot` is confused about the contract, and a
// plugin that reads it would be reading whatever the host happened to leave there.
pub fn takes_arg(op: text) -> boolean {
  return (op == OP_APPLY_OP) || (op == OP_MAKE_OP);
}

// ---------- request frames ----------
// { "op": text, "state": bytes, "arg": bytes }
//
// `state` is always present (the empty byte string for `initial_state`), so the plugin never has
// to distinguish "absent" from "empty" — an absent-vs-empty ambiguity across a wire is exactly
// the kind of thing that produces a plausible wrong answer instead of an error.
pub fn request(op: text, state_b64: text, arg_b64: text) -> vector<u8> {
  es: vector<CborEntry> = [];
  es += [ pa_te("op", pa_ctext(op)) ];
  es += [ pa_te("state", pa_cbytes(state_b64)) ];
  es += [ pa_te("arg", pa_cbytes(arg_b64)) ];
  return encode(CMap { entries: es });
}

pub fn req_op(frame: vector<u8>) -> text        { return pa_text(pa_decode(frame), "op"); }
pub fn req_state_b64(frame: vector<u8>) -> text { return pa_bytes(pa_decode(frame), "state"); }
pub fn req_arg_b64(frame: vector<u8>) -> text   { return pa_bytes(pa_decode(frame), "arg"); }

// ---------- reply frames ----------
// success: { "ok": true,  "out": bytes }
// failure: { "ok": false, "err": text }
//
// A failure carries a CODE, not a message: the host shows it to a person, and a free-form string
// from the plugin would be the one channel out of the sandbox that is neither bytes-the-host-
// inspects (render) nor a value it asked for. §9c.5 keeps render output the only rich channel.
pub fn reply_ok(out_b64: text) -> vector<u8> {
  es: vector<CborEntry> = [];
  es += [ pa_te("ok", pa_cbool(true)) ];
  es += [ pa_te("out", pa_cbytes(out_b64)) ];
  return encode(CMap { entries: es });
}

pub fn reply_err(code: text) -> vector<u8> {
  es: vector<CborEntry> = [];
  es += [ pa_te("ok", pa_cbool(false)) ];
  es += [ pa_te("err", pa_ctext(code)) ];
  return encode(CMap { entries: es });
}

// true only for a well-formed success reply. A frame that does not decode, or whose `ok` is not
// the boolean true, reads as failure — the fail-closed direction.
pub fn reply_is_ok(frame: vector<u8>) -> boolean {
  if !pa_decode_ok(frame) { return false; }
  match pa_get(pa_decode(frame), "ok") {
    CBool { value } => { return value; },
    _ => { return false; },
  }
}

pub fn reply_out_b64(frame: vector<u8>) -> text { return pa_bytes(pa_decode(frame), "out"); }
pub fn reply_err_code(frame: vector<u8>) -> text { return pa_text(pa_decode(frame), "err"); }

// ---------- the error codes a host may produce ----------
// Closed set, so a host and a UI cannot drift. `ERR_UNKNOWN_OP` is the version-skew case above;
// `ERR_MALFORMED` is a frame that did not decode; `ERR_PLUGIN` is the plugin reporting it could
// not satisfy a well-formed request (§9c.2 permits that — e.g. load_snapshot on foreign bytes).
pub const ERR_UNKNOWN_OP: text = "unknown-op";
pub const ERR_MALFORMED:  text = "malformed-frame";
pub const ERR_PLUGIN:     text = "plugin-error";

// Validate an inbound request frame the way a host must, BEFORE dispatching it: decode, then
// check the vocabulary. Returns "" when the frame is acceptable, else the error code to reply
// with. Centralised so every host does the same two checks in the same order rather than each
// re-deriving them — the §9c.2 contract has one front door.
pub fn check_request(frame: vector<u8>) -> text {
  if !pa_decode_ok(frame) { return ERR_MALFORMED; }
  op = pa_text(pa_decode(frame), "op");
  if !valid_op(op) { return ERR_UNKNOWN_OP; }
  return "";
}

tests/pluginabi.loft

// Tests for pluginabi — the §9c.2 plugin call protocol. Run: `loft test`.
//
// The load-bearing test is the last one: a COMPLETE six-operation plugin driven end to end over
// nothing but frames. If the protocol were missing something a real plugin needs, that test could
// not be written — which is the point of writing it here rather than asserting field-by-field.
use pluginabi;
use cbor;
use crypto;

// Fixture helpers. The library keeps its CBOR shims PRIVATE — exporting them is not a protocol
// library's job — so a test that builds its own payloads brings its own, exactly as a real plugin
// author would.
fn t_te(k: text, v: CborValue) -> CborEntry { return CborEntry { key: CText { value: k }, value: v }; }
fn t_cuint(n: integer) -> CborValue { return CInt { value: n }; }
fn t_decode(bytes: vector<u8>) -> CborValue { d = decode(bytes); v = d.value; return v; }
fn t_decode_ok(bytes: vector<u8>) -> boolean { d = decode(bytes); return d.ok; }
fn t_uint(m: CborValue, key: text) -> integer {
  match m {
    CMap { entries } => {
      for i in 0..len(entries) {
        e = entries[i] ?? CborEntry { key: CNull, value: CNull };
        match e.key { CText { value } => {
          if value == key { match e.value { CInt { value } => { return value; }, _ => {} } }
        }, _ => {} }
      }
    },
    _ => {},
  }
  return 0;
}

fn test_request_round_trip() {
  st = bytes_to_base64([1 as u8, 2 as u8, 255 as u8]);   // binary, incl. a non-UTF-8 byte
  ar = bytes_to_base64([0 as u8, 7 as u8]);
  f = request(OP_APPLY_OP, st, ar);

  assert(req_op(f) == OP_APPLY_OP, "op round-trips");
  assert(req_state_b64(f) == st, "state round-trips byte-exactly");
  assert(req_arg_b64(f) == ar, "arg round-trips byte-exactly");

  // the empty-state case: `initial_state` sends "" rather than omitting the field, so a plugin
  // never has to tell absent from empty
  e = request(OP_INITIAL_STATE, "", "");
  assert(req_state_b64(e) == "" && req_arg_b64(e) == "", "empty state/arg survive as empty");
  assert(req_op(e) == OP_INITIAL_STATE, "op survives alongside empty payloads");
}

fn test_reply_round_trip_and_fails_closed() {
  out = bytes_to_base64([9 as u8, 9 as u8]);
  ok = reply_ok(out);
  assert(reply_is_ok(ok), "a success reply reads as ok");
  assert(reply_out_b64(ok) == out, "its payload round-trips");

  er = reply_err(ERR_PLUGIN);
  assert(!reply_is_ok(er), "a failure reply does NOT read as ok");
  assert(reply_err_code(er) == ERR_PLUGIN, "its code round-trips");

  // FAIL-CLOSED: anything that is not a well-formed success must read as failure, never as ok.
  // This is the direction that matters — a garbled frame optimistically read as success would
  // hand the host whatever bytes happened to decode.
  assert(!reply_is_ok([]), "an empty frame is not ok");
  assert(!reply_is_ok([1 as u8, 2 as u8, 3 as u8]), "garbage bytes are not ok");
  assert(!reply_is_ok(request(OP_RENDER, "", "")), "a REQUEST frame is not a success reply");
}

// Large-payload coverage (loft#651). Every other test used 2-3 byte payloads; a real `merge3`
// frame carries a whole serialized document, so large fields are the COMMON case. This asserts the
// contract holds to 300 bytes. NOTE: loft#651 (the crash where `pa_get`'s old statement-form
// `match` corrupted a heap CborValue past ~54 bytes) is CONTEXT-SENSITIVE — it fires reliably as a
// standalone program but NOT inside this test harness, so this test does not by itself distinguish
// the buggy `pa_get` from the fixed one. The reliable repro is a program (see the zero-trust repo's
// upstream/match-stmt-return-heap-repro.loft); this test guards the contract and documents intent.
fn test_large_payload_round_trip() {
  for sz in [1, 40, 53, 54, 100, 300] {
    raw: vector<u8> = []; for i in 0..sz { raw += [ ((i * 7 + 3) & 255) as u8 ]; }
    b64 = bytes_to_base64(raw);
    // request: state AND arg both large; both must survive the read path (arg is the LAST field).
    f = request(OP_APPLY_OP, b64, b64);
    assert(req_state_b64(f) == b64, "large state round-trips byte-exactly (sz={sz})");
    assert(req_arg_b64(f) == b64, "large arg round-trips byte-exactly (sz={sz})");
    // reply: a large `out` must survive reply_out_b64, and reply_is_ok must not choke on it.
    r = reply_ok(b64);
    assert(reply_is_ok(r), "a large success reply reads as ok (sz={sz})");
    assert(reply_out_b64(r) == b64, "large reply out round-trips byte-exactly (sz={sz})");
  }
  println("plugins: pluginabi — large frame payloads (to 300 bytes) round-trip byte-exactly");
}

fn test_check_request_is_the_front_door() {
  good = request(OP_SNAPSHOT, "", "");
  assert(check_request(good) == "", "a well-formed known op passes");

  assert(check_request([7 as u8, 7 as u8]) == ERR_MALFORMED, "undecodable -> malformed");
  assert(check_request([]) == ERR_MALFORMED, "empty -> malformed");

  // version skew: an op this host does not know must be REJECTED, not guessed at
  unknown = request("apply_ops", "", "");            // note the typo/newer name
  assert(check_request(unknown) == ERR_UNKNOWN_OP, "an unknown op is rejected, not interpreted");

  // the vocabulary is exactly six
  n = 0;
  for o in [OP_INITIAL_STATE, OP_APPLY_OP, OP_MAKE_OP, OP_RENDER, OP_SNAPSHOT, OP_LOAD_SNAPSHOT] {
    if valid_op(o) { n = n + 1; }
  }
  assert(n == 6, "all six operations are in the vocabulary");
  assert(!valid_op("") && !valid_op("eval") && !valid_op("exec"), "nothing else is");
  assert(takes_arg(OP_APPLY_OP) && takes_arg(OP_MAKE_OP), "apply_op/make_op read arg");
  assert(!takes_arg(OP_SNAPSHOT) && !takes_arg(OP_INITIAL_STATE), "the others do not");
}

// ---------- a complete plugin, driven only by frames ----------
// A counter: state is CBOR { "n": uint }; an op is CBOR { "d": uint } (a delta). The plugin owns
// that encoding — the protocol only carries bytes, which is the §9c.2 contract.
fn plug_state(n: integer) -> text {
  es: vector<CborEntry> = []; es += [ t_te("n", t_cuint(n)) ];
  return bytes_to_base64(encode(CMap { entries: es }));
}
fn plug_n(state_b64: text) -> integer {
  if state_b64 == "" { return 0; }
  return t_uint(t_decode(base64_to_bytes(state_b64)), "n");
}
fn plug_delta(arg_b64: text) -> integer {
  if arg_b64 == "" { return 0; }
  return t_uint(t_decode(base64_to_bytes(arg_b64)), "d");
}

// THE PLUGIN: one entry point, frame in -> frame out. This is what the wasm pump calls.
fn plugin_dispatch(frame: vector<u8>) -> vector<u8> {
  bad = check_request(frame);
  if bad != "" { return reply_err(bad); }
  op = req_op(frame);
  st = req_state_b64(frame);
  ar = req_arg_b64(frame);

  if op == OP_INITIAL_STATE { return reply_ok(plug_state(0)); }
  if op == OP_APPLY_OP      { return reply_ok(plug_state(plug_n(st) + plug_delta(ar))); }
  if op == OP_MAKE_OP       { return reply_ok(ar); }
  if op == OP_RENDER        { return reply_ok(base64_encode("count={plug_n(st)}")); }
  if op == OP_SNAPSHOT      { return reply_ok(st); }
  if op == OP_LOAD_SNAPSHOT {
    if !t_decode_ok(base64_to_bytes(st)) { return reply_err(ERR_PLUGIN); }   // foreign bytes
    return reply_ok(st);
  }
  return reply_err(ERR_UNKNOWN_OP);
}

fn call(op: text, st: text, ar: text) -> vector<u8> { return plugin_dispatch(request(op, st, ar)); }

fn test_a_whole_plugin_over_frames() {
  // initial_state
  r = call(OP_INITIAL_STATE, "", "");
  assert(reply_is_ok(r), "initial_state succeeds");
  st = reply_out_b64(r);
  assert(plug_n(st) == 0, "a fresh state counts zero");

  // make_op then apply_op, twice — the host never touches the plugin's encoding
  d5 = reply_out_b64(call(OP_MAKE_OP, st, plug_delta_arg(5)));
  st = reply_out_b64(call(OP_APPLY_OP, st, d5));
  d7 = reply_out_b64(call(OP_MAKE_OP, st, plug_delta_arg(7)));
  st = reply_out_b64(call(OP_APPLY_OP, st, d7));
  assert(plug_n(st) == 12, "5 + 7 folded through frames = 12");

  // render — the host-visible channel
  rr = call(OP_RENDER, st, "");
  assert(reply_is_ok(rr) && base64_decode(reply_out_b64(rr)) == "count=12", "render reflects state");

  // snapshot -> load_snapshot round-trips to the SAME state (the §9c.1 rollover contract)
  snap = reply_out_b64(call(OP_SNAPSHOT, st, ""));
  back = call(OP_LOAD_SNAPSHOT, snap, "");
  assert(reply_is_ok(back), "a snapshot loads");
  assert(plug_n(reply_out_b64(back)) == 12, "and restores the same count");

  // a plugin may refuse a well-formed request — foreign snapshot bytes
  bad = call(OP_LOAD_SNAPSHOT, bytes_to_base64([200 as u8, 201 as u8]), "");
  assert(!reply_is_ok(bad) && reply_err_code(bad) == ERR_PLUGIN,
    "load_snapshot on foreign bytes is a plugin error, not a protocol error");

  // and the host rejects an op the plugin never sees
  sk = plugin_dispatch(request("teleport", st, ""));
  assert(!reply_is_ok(sk) && reply_err_code(sk) == ERR_UNKNOWN_OP, "an unknown op never reaches the plugin body");

  println("plugins: pluginabi — §9c.2 frame protocol + a whole plugin driven over frames OK");
}

fn plug_delta_arg(d: integer) -> text {
  es: vector<CborEntry> = []; es += [ t_te("d", t_cuint(d)) ];
  return bytes_to_base64(encode(CMap { entries: es }));
}
