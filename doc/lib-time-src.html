src/time.loft

// Copyright (c) 2026 Jurjen Stellingwerff
// SPDX-License-Identifier: LGPL-3.0-or-later
//
// time — basic date/time operations for loft.
//
// A point in time is a plain `integer` (i64) holding milliseconds since
// the Unix epoch (1970-01-01T00:00:00 UTC) — the same unit `now()`
// returns and the same model as JavaScript's `Date`.  All calendar math
// uses the proleptic Gregorian calendar (Howard Hinnant's
// days_from_civil / civil_from_days), so results match `js_sys::Date`
// field-for-field; the library is pure loft and runs identically on the
// interpreter, `--native`, and wasm.
//
// Timezone model: formatting and field extraction are in UTC.  For
// local-day bucketing pass a fixed UTC offset in minutes
// (`local_day` / `to_local` / `today`).  There is NO timezone database
// and NO daylight-saving handling — a fixed offset only.
//
// Public API:
//   Construct/parse: from_ymd, from_millis, parse, today
//   Fields:          year, month, day, hour, minute, second,
//                    weekday (0=Mon..6=Sun), iso_year, iso_week
//   Step:            add_days, add_weeks, add_seconds
//   Difference:      days_between, seconds_between
//   Boundaries:      start_of_day, start_of_week (Monday)
//   Local:           to_local, local_day, today
//   Format:          format_date, format_time, format_seconds,
//                    format_datetime, format_iso, weekday_name,
//                    month_name
//   Parse failure returns null (i64::MIN) — test with `!result`.

const MS_PER_DAY    = 86400000;
const MS_PER_HOUR   = 3600000;
const MS_PER_MINUTE = 60000;
const MS_PER_SECOND = 1000;

// ── floor division / modulo (correct for pre-1970 negatives) ──
// loft `/` and `%` truncate toward zero; the calendar math needs floor.

fn floor_div(fa: integer, fb: integer) -> integer {
    q = fa / fb ?? 0;
    r = fa % fb ?? 0;
    if (r != 0) && ((fa < 0) != (fb < 0)) { q - 1 } else { q }
}

// Days since the Unix epoch for this instant (floored to the UTC day).
fn epoch_day(ed_t: integer) -> integer { floor_div(ed_t, MS_PER_DAY) }

// ── civil calendar (Hinnant) ──────────────────────────────────

// (year, month, day) for a day number `z` counted from 1970-01-01.
fn civil_from_days(z: integer) -> (integer, integer, integer) {
    zz  = z + 719468;
    eshift = if zz >= 0 { zz } else { zz - 146096 };
    era = floor_div(eshift, 146097);
    doe = zz - era * 146097;                                  // [0, 146096]
    yoe = (doe - doe / 1460 + doe / 36524 - doe / 146096) / 365;  // [0, 399]
    y   = yoe + era * 400;
    doy = doe - (365 * yoe + yoe / 4 - yoe / 100);            // [0, 365]
    mp  = (5 * doy + 2) / 153;                                // [0, 11]
    d   = doy - (153 * mp + 2) / 5 + 1;                       // [1, 31]
    m   = if mp < 10 { mp + 3 } else { mp - 9 };              // [1, 12]
    yr  = if m <= 2 { y + 1 } else { y };
    (yr, m, d)
}

// Day number counted from 1970-01-01 for a (year, month, day).
fn days_from_civil(cy: integer, cm: integer, cd: integer) -> integer {
    yy  = if cm <= 2 { cy - 1 } else { cy };
    yshift = if yy >= 0 { yy } else { yy - 399 };
    era = floor_div(yshift, 400);
    yoe = yy - era * 400;                                     // [0, 399]
    mpart = if cm > 2 { cm - 3 } else { cm + 9 };
    doy = (153 * mpart + 2) / 5 + cd - 1;                     // [0, 365]
    doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;              // [0, 146096]
    era * 146097 + doe - 719468
}

// ── construction / parsing ────────────────────────────────────

// Epoch-ms at UTC midnight of the given calendar date.
pub fn from_ymd(fy: integer, fmo: integer, fd: integer) -> integer {
    days_from_civil(fy, fmo, fd) * MS_PER_DAY
}

// Identity wrapper — for readability where a raw ms count becomes a time.
pub fn from_millis(ms: integer) -> integer { ms }

// Epoch-SECONDS boundary.  Garmin / Strava REST APIs speak Unix seconds
// (`.timestamp()`), while loft times are milliseconds — convert at the edge.
pub fn from_unix_seconds(secs: integer) -> integer { secs * MS_PER_SECOND }
pub fn to_unix_seconds(uts: integer) -> integer { floor_div(uts, MS_PER_SECOND) }

fn digit_value(dc: character) -> integer {
    if dc >= '0' && dc <= '9' { (dc as integer) - ('0' as integer) } else { -1 }
}

// Parse `count` decimal digits beginning at character index `start`.
// Returns -1 if any character in the span is not a digit.
fn digits_at(ds: text, start: integer, count: integer) -> integer {
    val  = 0;
    seen = 0;
    for dch in ds {
        if dch#index >= start && dch#index < start + count {
            dv = digit_value(dch);
            if dv < 0 { return -1; }
            val = val * 10 + dv;
            seen += 1;
        }
    }
    if seen == count { val } else { -1 }
}

// Parse an ISO date "YYYY-MM-DD", optionally followed by " HH:MM[:SS]"
// or "THH:MM[:SS]" with an optional trailing "Z".  Returns null on any
// malformed input (test with `!result`).
pub fn parse(ps: text) -> integer? {
    n = size(ps);
    if n < 10 { return null; }
    py  = digits_at(ps, 0, 4);
    pmo = digits_at(ps, 5, 2);
    pd  = digits_at(ps, 8, 2);
    if py < 0 || pmo < 0 || pd < 0 { return null; }
    pt = from_ymd(py, pmo, pd);
    if n >= 16 {
        ph = digits_at(ps, 11, 2);
        pmi = digits_at(ps, 14, 2);
        if ph >= 0 && pmi >= 0 {
            pt += ph * MS_PER_HOUR + pmi * MS_PER_MINUTE;
            if n >= 19 {
                psec = digits_at(ps, 17, 2);
                if psec >= 0 { pt += psec * MS_PER_SECOND; }
            }
        }
    }
    pt
}

// ── field accessors ───────────────────────────────────────────

pub fn year(yt: integer) -> integer {
    yc = civil_from_days(epoch_day(yt));
    yc.0
}

pub fn month(mt: integer) -> integer {
    mc = civil_from_days(epoch_day(mt));
    mc.1
}

pub fn day(dt: integer) -> integer {
    dc = civil_from_days(epoch_day(dt));
    dc.2
}

fn ms_of_day(mt2: integer) -> integer { floor_mod(mt2, MS_PER_DAY) ?? 0 }

pub fn hour(ht: integer)   -> integer { ms_of_day(ht) / MS_PER_HOUR }
pub fn minute(mit: integer) -> integer { (ms_of_day(mit) / MS_PER_MINUTE) % 60 }
pub fn second(st: integer)  -> integer { (ms_of_day(st) / MS_PER_SECOND) % 60 }

// 0 = Monday .. 6 = Sunday.  1970-01-01 was a Thursday (weekday 3).
pub fn weekday(wt: integer) -> integer { floor_mod(epoch_day(wt) + 3, 7) ?? 0 }

// ── stepping ──────────────────────────────────────────────────

pub fn add_days(adt: integer, days: integer)     -> integer { adt + days * MS_PER_DAY }
pub fn add_weeks(awt: integer, weeks: integer)   -> integer { awt + weeks * 7 * MS_PER_DAY }
pub fn add_seconds(ast: integer, secs: integer)  -> integer { ast + secs * MS_PER_SECOND }

// ── differences ───────────────────────────────────────────────

// Whole calendar days from `a` to `b` (b later → positive).
pub fn days_between(da: integer, db: integer) -> integer {
    epoch_day(db) - epoch_day(da)
}

pub fn seconds_between(sa: integer, sb: integer) -> integer {
    (sb - sa) / MS_PER_SECOND
}

// ── boundaries ────────────────────────────────────────────────

pub fn start_of_day(sdt: integer) -> integer { epoch_day(sdt) * MS_PER_DAY }

// Monday 00:00 UTC of the week containing `t`.
pub fn start_of_week(swt: integer) -> integer {
    start_of_day(swt) - weekday(swt) * MS_PER_DAY
}

// Combine a date with a time-of-day given as components (datetime.combine).
pub fn at_time(atd: integer, hh: integer, mm: integer, ss: integer) -> integer {
    start_of_day(atd) + hh * MS_PER_HOUR + mm * MS_PER_MINUTE + ss * MS_PER_SECOND
}

// Combine a date with an "HH:MM" or "HH:MM:SS" time string.  Mirrors
// `datetime.combine(date, time.fromisoformat(s))`.  Null on a malformed time.
pub fn combine(cmd: integer, time_text: text) -> integer? {
    cn = size(time_text);
    if cn < 5 { return null; }
    chh = digits_at(time_text, 0, 2);
    cmm = digits_at(time_text, 3, 2);
    if chh < 0 || cmm < 0 { return null; }
    css = 0;
    if cn >= 8 {
        css = digits_at(time_text, 6, 2);
        if css < 0 { css = 0; }
    }
    at_time(cmd, chh, cmm, css)
}

// ── ISO week numbering ────────────────────────────────────────

// (iso_year, iso_week) — weeks start Monday; week 1 contains the year's
// first Thursday.
fn iso_calendar(it: integer) -> (integer, integer) {
    iday = epoch_day(it);
    iwd  = weekday(it);                       // 0=Mon..6=Sun
    thursday = iday - iwd + 3;                // Thursday of this week
    tc = civil_from_days(thursday);
    iy = tc.0;
    jan1 = days_from_civil(iy, 1, 1);
    week = (thursday - jan1) / 7 + 1;
    (iy, week)
}

pub fn iso_year(iyt: integer) -> integer {
    iyc = iso_calendar(iyt);
    iyc.0
}

pub fn iso_week(iwt: integer) -> integer {
    iwc = iso_calendar(iwt);
    iwc.1
}

// ── local time (fixed offset, no DST) ─────────────────────────

// Shift so that the field accessors / formatters yield local values.
pub fn to_local(lt: integer, offset_min: integer) -> integer {
    lt + offset_min * MS_PER_MINUTE
}

// UTC-midnight key of the local calendar day containing `t`.
pub fn local_day(ldt: integer, offset_min: integer) -> integer {
    start_of_day(to_local(ldt, offset_min))
}

// The local calendar day for "now" at the given offset.
pub fn today(offset_min: integer) -> integer {
    local_day(now(), offset_min)
}

// ── formatting ────────────────────────────────────────────────

pub fn format_date(fdt: integer) -> text {
    fc = civil_from_days(epoch_day(fdt));
    fyy = fc.0;
    fmm = fc.1;
    fdd = fc.2;
    "{fyy:04}-{fmm:02}-{fdd:02}"
}

pub fn format_time(ftt: integer) -> text {
    fth = hour(ftt);
    ftm = minute(ftt);
    "{fth:02}:{ftm:02}"
}

pub fn format_seconds(fst: integer) -> text {
    fsh = hour(fst);
    fsm = minute(fst);
    fss = second(fst);
    "{fsh:02}:{fsm:02}:{fss:02}"
}

pub fn format_datetime(fdtt: integer) -> text {
    fdd2 = format_date(fdtt);
    ftt2 = format_time(fdtt);
    "{fdd2} {ftt2}"
}

pub fn format_iso(fit: integer) -> text {
    fid = format_date(fit);
    fis = format_seconds(fit);
    "{fid}T{fis}Z"
}

pub fn weekday_name(wnt: integer) -> text {
    match weekday(wnt) {
        0 => "Mon",
        1 => "Tue",
        2 => "Wed",
        3 => "Thu",
        4 => "Fri",
        5 => "Sat",
        _ => "Sun",
    }
}

pub fn month_name(mnt: integer) -> text {
    match month(mnt) {
        1  => "Jan",
        2  => "Feb",
        3  => "Mar",
        4  => "Apr",
        5  => "May",
        6  => "Jun",
        7  => "Jul",
        8  => "Aug",
        9  => "Sep",
        10 => "Oct",
        11 => "Nov",
        _  => "Dec",
    }
}

// ══════════════════════════════════════════════════════════════
//  DateTime / Duration — first-grade value types (0.2.0)
// ══════════════════════════════════════════════════════════════
//
// `DateTime` and `Duration` are zero-cost `value struct`s wrapping the same
// epoch-millisecond `integer` the functions above use — so they add NO heap
// allocation as a record field or vector element (a `vector<DateTime>` costs
// the same as `vector<integer>`), yet they are a DISTINCT type: `dt + 5` is a
// compile error, only `dt + hours(2)` steps a time.  They layer on top of the
// integer API above (which stays exactly as it was — the 0.1.0 surface still
// works); pick whichever fits the call site.
//
//   Construct:  date(y,mo,d) · datetime(y,mo,d,h,mi,s) · ms as DateTime ·
//               "2026-07-08T12:00:00Z" as DateTime · now() as DateTime
//   Read:       dt.year() … dt.second(), dt.weekday(), dt.weekday_name(), …
//   Compare:    dt1 < dt2 · dt1 == dt2  (all six)
//   Span:       dt2 - dt1 -> Duration ;  dt + hours(2) -> DateTime
//   Format:     "{dt}" (YYYY-MM-DD HH:MM) · "{dt:date}" · "{dt:iso}" · "{dt:time}"
//
// FALLIBILITY: a `value struct` has no null, so a fallible parse cannot return
// a `DateTime` AND signal failure.  Keep failure at the integer level — `parse`
// above returns null on bad input — then wrap the checked value:
//     ms = parse(s); if !ms { …handle… } else { dt = ms as DateTime }
// The `"…" as DateTime` conversion is a TOTAL best-effort parse (bad input →
// epoch) for the ergonomic path where an invalid date need not be caught.

pub value struct DateTime { ms: integer }
pub value struct Duration { ms: integer }

// ── DateTime construction ──
pub fn date(dcy: integer, dcmo: integer, dcd: integer) -> DateTime {

}

                tch: integer, tcmi: integer, tcs: integer) -> DateTime {

}

// ── custom formatting (the per-type to_text hook) ──
// "" / "datetime" -> YYYY-MM-DD HH:MM · "date" -> YYYY-MM-DD · "time" -> HH:MM
// "seconds" -> HH:MM:SS · "iso" -> …THH:MM:SSZ · "wday" -> Mon · "month" -> May
pub fn to_text(self: DateTime, spec: text) -> text {
    // The if-chain is bound to `r` and returned, NOT left as the tail expression:
    // a tail `if` inside a format-hook `to_text(self, spec)` mis-selects its branch
    // on both backends (a loft format-hook bug — repro filed upstream). Binding the
    // result to a local sidesteps it.
    // Each branch is wrapped in "{…}" so every arm is an owned text: on native a
    // bare `text`-returning call whose body is a match of string literals (weekday_name /
    // month_name → &str) does NOT unify with a String-returning arm (format_* interpolate),
    // giving rustc E0308. Interpolating each arm normalises them (a second loft bug, filed).
    r = if spec == "date" { "{format_date(self.ms)}" }
        else if spec == "time" { "{format_time(self.ms)}" }
        else if spec == "seconds" || spec == "secs" { "{format_seconds(self.ms)}" }
        else if spec == "iso" { "{format_iso(self.ms)}" }
        else if spec == "wday" || spec == "weekday" { "{weekday_name(self.ms)}" }
        else if spec == "month" { "{month_name(self.ms)}" }
        else { "{format_datetime(self.ms)}" };
    r
}
// A Duration renders as [-]H:MM:SS (hours may exceed 24).
pub fn to_text(self: Duration, _spec: text) -> text {
    neg = self.ms < 0;
    tot = if neg { 0 - self.ms } else { self.ms };
    ss = (tot / MS_PER_SECOND) % 60;
    mm = (tot / MS_PER_MINUTE) % 60;
    hh = tot / MS_PER_HOUR;
    sign = if neg { "-" } else { "" };
    "{sign}{hh}:{mm:02}:{ss:02}"
}

fn OpConvIntFromDateTime(civ: DateTime) -> integer { civ.ms }
fn OpConvDateTimeFromInt(civ: integer)  -> DateTime { DateTime { ms: civ } }
// text -> DateTime: a TOTAL best-effort parse (bad input -> epoch); see the
// FALLIBILITY note above for the checked path via `parse`.
fn OpConvDateTimeFromText(civ: text) -> DateTime {
    DateTime { ms: parse(civ) ?? 0 }
}

// Copyright (c) 2026 Jurjen Stellingwerff
// SPDX-License-Identifier: LGPL-3.0-or-later
//
// Basic date/time operation tests — run on interpreter, --native, and
// wasm.  All goldens are hand-computed against the proleptic Gregorian
// calendar so they pin cross-backend parity.
use time;

fn test_parse_and_fields() {
    t = (time::parse("2026-05-25") ?? 0);
    assert(time::year(t) == 2026, "year {time::year(t)}");
    assert(time::month(t) == 5, "month {time::month(t)}");
    assert(time::day(t) == 25, "day {time::day(t)}");
    // 2026-05-25 is a Monday.
    assert(time::weekday(t) == 0, "weekday {time::weekday(t)}");
    assert(time::weekday_name(t) == "Mon", "wname {time::weekday_name(t)}");
    assert(time::month_name(t) == "May", "mname {time::month_name(t)}");
}

fn test_parse_datetime() {
    t = (time::parse("2026-05-25 14:30:07") ?? 0);
    assert(time::hour(t) == 14, "hour {time::hour(t)}");
    assert(time::minute(t) == 30, "minute {time::minute(t)}");
    assert(time::second(t) == 7, "second {time::second(t)}");
    iso = (time::parse("2026-05-25T14:30:07Z") ?? 0);
    assert(iso == t, "T/Z form equals space form");
}

fn test_parse_failure() {
    bad = time::parse("not-a-date");
    assert(!bad, "garbage parses to null");
    short = time::parse("2026-05");
    assert(!short, "too-short parses to null");
}

fn test_roundtrip_format() {
    t = time::from_ymd(1999, 12, 31);
    assert(time::format_date(t) == "1999-12-31", "fmt {time::format_date(t)}");
    dt = (time::parse("2000-01-01 00:00:00") ?? 0);
    assert(time::format_iso(dt) == "2000-01-01T00:00:00Z", "iso {time::format_iso(dt)}");
    assert(time::format_datetime(dt) == "2000-01-01 00:00", "dtfmt {time::format_datetime(dt)}");
}

fn test_arithmetic() {
    t = (time::parse("2026-01-01") ?? 0);
    wk = time::add_weeks(t, 1);
    assert(time::format_date(wk) == "2026-01-08", "add_weeks {time::format_date(wk)}");
    dd = time::add_days(t, 365);
    assert(time::format_date(dd) == "2027-01-01", "add 365 {time::format_date(dd)}");
    // leap-year boundary: 2024-02-28 + 1 day = 2024-02-29
    leap = time::add_days((time::parse("2024-02-28") ?? 0), 1);
    assert(time::format_date(leap) == "2024-02-29", "leap {time::format_date(leap)}");
}

fn test_differences() {
    a = (time::parse("2026-01-01") ?? 0);
    b = (time::parse("2026-12-31") ?? 0);
    assert(time::days_between(a, b) == 364, "days_between {time::days_between(a, b)}");
    s = time::seconds_between((time::parse("2026-01-01 00:00:00") ?? 0),
                              (time::parse("2026-01-01 01:00:00") ?? 0));
    assert(s == 3600, "seconds_between {s}");
}

fn test_boundaries() {
    t = (time::parse("2026-05-25 14:30:07") ?? 0);
    sod = time::start_of_day(t);
    assert(time::format_iso(sod) == "2026-05-25T00:00:00Z", "start_of_day {time::format_iso(sod)}");
    // 2026-05-25 is a Monday, so start_of_week == start_of_day.
    sow = time::start_of_week(t);
    assert(time::format_date(sow) == "2026-05-25", "start_of_week {time::format_date(sow)}");
    // Wednesday 2026-05-27 -> week start Monday 2026-05-25.
    wed = (time::parse("2026-05-27") ?? 0);
    assert(time::format_date(time::start_of_week(wed)) == "2026-05-25",
           "wed week start {time::format_date(time::start_of_week(wed))}");
}

fn test_iso_week() {
    // 2026-01-01 is a Thursday -> ISO week 1 of 2026.
    a = (time::parse("2026-01-01") ?? 0);
    assert(time::iso_year(a) == 2026, "iso_year {time::iso_year(a)}");
    assert(time::iso_week(a) == 1, "iso_week {time::iso_week(a)}");
    // 2021-01-01 is a Friday -> belongs to ISO week 53 of 2020.
    b = (time::parse("2021-01-01") ?? 0);
    assert(time::iso_year(b) == 2020, "iso_year2 {time::iso_year(b)}");
    assert(time::iso_week(b) == 53, "iso_week2 {time::iso_week(b)}");
}

fn test_local_offset() {
    // 23:30 UTC at +120 min becomes 01:30 the next local day.
    t = (time::parse("2026-05-25 23:30:00") ?? 0);
    loc = time::to_local(t, 120);
    assert(time::format_datetime(loc) == "2026-05-26 01:30", "local {time::format_datetime(loc)}");
    assert(time::format_date(time::local_day(t, 120)) == "2026-05-26",
           "local_day {time::format_date(time::local_day(t, 120))}");
}

fn test_epoch_seconds() {
    // Garmin/Strava boundary: Unix seconds <-> loft ms.
    secs = 1748000000;
    t = time::from_unix_seconds(secs);
    assert(t == 1748000000000, "from_unix_seconds {t}");
    assert(time::to_unix_seconds(t) == secs, "to_unix_seconds {time::to_unix_seconds(t)}");
    // round-trip through a parsed datetime.
    dt = (time::parse("2026-05-25 14:30:07") ?? 0);
    assert(time::from_unix_seconds(time::to_unix_seconds(dt)) == dt, "secs round-trip");
}

fn test_combine() {
    d = (time::parse("2026-05-25") ?? 0);
    e = (time::combine(d, "14:30") ?? 0);
    assert(time::format_datetime(e) == "2026-05-25 14:30", "combine HH:MM {time::format_datetime(e)}");
    e2 = (time::combine(d, "09:05:42") ?? 0);
    assert(time::format_seconds(e2) == "09:05:42", "combine HH:MM:SS {time::format_seconds(e2)}");
    assert(time::format_date(e2) == "2026-05-25", "combine keeps date");
    at = time::at_time(d, 23, 59, 0);
    assert(time::format_datetime(at) == "2026-05-25 23:59", "at_time {time::format_datetime(at)}");
    bad = time::combine(d, "xx");
    assert(!bad, "combine bad time -> null");
}

fn main() {
    test_epoch_seconds();
    test_combine();
    test_parse_and_fields();
    test_parse_datetime();
    test_parse_failure();
    test_roundtrip_format();
    test_arithmetic();
    test_differences();
    test_boundaries();
    test_iso_week();
    test_local_offset();
    println("all time tests passed");
}

// DateTime / Duration value-type tests (0.2.0).  Goldens hand-computed
// against the proleptic Gregorian calendar; every case must agree on the
// interpreter and --native (and wasm).
use time;

fn test_construct_and_epoch() {
    // absolute anchors — catch a systematic offset the round-trips would hide
    e0 = time::datetime(1970, 1, 1, 0, 0, 0);
    assert((e0 as integer) == 0, "unix epoch is ms 0");
    y2k = time::datetime(2000, 1, 1, 0, 0, 0);
    assert((y2k as integer) == 946684800000, "2000-01-01 known epoch ms");
    assert((time::date(2000, 1, 1) as integer) == 946684800000, "date() midnight matches");
}

fn test_fields() {
    d = time::datetime(2026, 7, 8, 13, 30, 7);
    assert(d.year() == 2026,  "year {d.year()}");
    assert(d.month() == 7,    "month {d.month()}");
    assert(d.day() == 8,      "day {d.day()}");
    assert(d.hour() == 13,    "hour {d.hour()}");
    assert(d.minute() == 30,  "minute {d.minute()}");
    assert(d.second() == 7,   "second {d.second()}");
    // 2026-05-25 is a Monday (weekday 0)
    mon = time::date(2026, 5, 25);
    assert(mon.weekday() == 0, "weekday {mon.weekday()}");
    assert(mon.weekday_name() == "Mon", "wname {mon.weekday_name()}");
    assert(mon.month_name() == "May", "mname {mon.month_name()}");
}

fn test_format() {
    d = time::datetime(2026, 7, 8, 13, 30, 7);
    assert("{d:date}" == "2026-07-08", "date {d:date}");
    assert("{d:time}" == "13:30", "time {d:time}");
    assert("{d:seconds}" == "13:30:07", "secs {d:seconds}");
    assert("{d:iso}" == "2026-07-08T13:30:07Z", "iso {d:iso}");
    assert("{d}" == "2026-07-08 13:30", "bare {d}");
    assert("{d:month}" == "Jul", "month spec {d:month}");
}

fn test_compare() {
    d1 = time::datetime(2026, 7, 8, 12, 0, 0);
    d2 = time::datetime(2026, 7, 8, 13, 30, 0);
    assert(d1 < d2, "op <");
    assert(d2 > d1, "op >");
    assert(d1 <= d1, "op <=");
    assert(d1 >= d1, "op >=");
    assert(d1 == d1, "op ==");
    assert(d1 != d2, "op !=");
}

fn test_arithmetic() {
    d1 = time::datetime(2026, 7, 8, 12, 0, 0);
    d2 = time::datetime(2026, 7, 8, 13, 30, 0);
    diff = d2 - d1;                     // DateTime - DateTime -> Duration
    assert(diff == time::minutes(90), "diff is 90 minutes");
    assert(diff.total_minutes() == 90, "total_minutes {diff.total_minutes()}");
    assert(diff.total_seconds() == 5400, "total_seconds {diff.total_seconds()}");
    d3 = d1 + time::minutes(90);         // DateTime + Duration -> DateTime
    assert(d3 == d2, "add duration reaches the later time");
    back = d1.minus(time::minutes(90));  // subtract a Duration (method)
    assert(back == time::datetime(2026, 7, 8, 10, 30, 0), "minus a duration");
}

fn test_duration() {
    assert(time::hours(1) < time::hours(2), "duration compare");
    assert((time::hours(1) + time::minutes(30)).total_minutes() == 90, "add durations");
    assert(time::days(1).total_hours() == 24, "day is 24h");
    assert(time::weeks(1).total_days() == 7, "week is 7d");
    assert((time::hours(2) * 3) == time::hours(6), "scale duration");
    assert(time::minutes(90).negate().total_minutes() == -90, "negate");
    assert("{time::minutes(90)}" == "1:30:00", "dur fmt {time::minutes(90)}");
    assert("{time::hours(25)}" == "25:00:00", "dur >24h");
    assert("{time::minutes(-90)}" == "-1:30:00", "negative dur fmt");
}

fn test_conversions() {
    d1 = time::datetime(2026, 7, 8, 12, 0, 0);
    epoch = d1 as integer;              // DateTime -> integer
    back = epoch as DateTime;           // integer  -> DateTime
    assert(back == d1, "int round-trip");
    parsed = "2026-07-08" as DateTime;  // text -> DateTime (total)
    assert(parsed == time::date(2026, 7, 8), "parse-on-cast date");
    parsed2 = "2026-07-08T13:30:00" as DateTime;
    assert(parsed2 == time::datetime(2026, 7, 8, 13, 30, 0), "parse-on-cast datetime");
}

fn test_now_and_vector() {
    // now() is a real wall-clock DateTime (via the total int->DateTime cast)
    cur = now() as DateTime;
    assert(cur > time::datetime(2020, 1, 1, 0, 0, 0), "now is after 2020");
    // value structs are zero-cost as vector elements — read one back out
    events = [time::datetime(2026, 1, 1, 0, 0, 0),
              time::datetime(2026, 6, 15, 9, 0, 0),
              time::datetime(2026, 12, 31, 23, 59, 59)];
    assert(events[1].month() == 6, "vector element field read");
    assert("{events[2]:date}" == "2026-12-31", "vector element format");
    // copy semantics: mutating a local copy leaves the source untouched
    first = events[0];
    first = first + time::days(10);
    assert(events[0].day() == 1, "value-struct copy: source element unchanged");
}

fn test_iso_and_millis() {
    d = time::datetime(2026, 5, 25, 8, 0, 0);   // Monday
    // the value-struct methods delegate to the integer API — cross-check on the
    // same instant (01-basics pins the integer fns to hand-goldens)
    assert(d.iso_year() == time::iso_year(d as integer), "iso_year delegates");
    assert(d.iso_week() == time::iso_week(d as integer), "iso_week delegates");
    assert(d.iso_week() >= 1 && d.iso_week() <= 53, "iso_week range {d.iso_week()}");
    assert(d.to_millis() == (d as integer), "to_millis == as integer");
}

fn test_duration_full() {
    assert((time::hours(2) - time::minutes(30)) == time::minutes(90), "duration subtract");
    assert(time::hours(1) <= time::hours(1), "dur <=");
    assert(time::hours(2) >= time::hours(1), "dur >=");
    assert(time::hours(2) > time::hours(1), "dur >");
    assert(time::hours(1) != time::hours(2), "dur !=");
    assert(time::minutes(90).total_millis() == 5400000, "total_millis");
}

fn test_format_aliases() {
    mon = time::date(2026, 5, 25);   // Monday
    assert("{mon:wday}" == "Mon", "wday spec {mon:wday}");
    assert("{mon:weekday}" == "Mon", "weekday alias {mon:weekday}");
    d = time::datetime(2026, 7, 8, 13, 30, 7);
    assert("{d:secs}" == "13:30:07", "secs alias {d:secs}");
}

fn test_edge_cases() {
    // pre-1970 (negative epoch ms) still fields + formats correctly
    old = time::datetime(1960, 6, 15, 12, 0, 0);
    assert(old.year() == 1960, "pre-1970 year {old.year()}");
    assert(old.month() == 6, "pre-1970 month");
    assert(old.day() == 15, "pre-1970 day");
    assert((old as integer) < 0, "pre-1970 is negative ms");
    assert("{old:date}" == "1960-06-15", "pre-1970 date fmt {old:date}");
    // garbage text -> DateTime is a TOTAL best-effort cast (epoch 0); the fallible
    // integer parse is the checked path
    assert(("garbage" as DateTime).to_millis() == 0, "total cast of garbage -> epoch");
    assert(!time::parse("garbage"), "checked parse returns null on garbage");
}
