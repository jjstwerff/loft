// Copyright (c) 2026 Jurjen Stellingwerff
// SPDX-License-Identifier: LGPL-3.0-or-later
// formcensus — THE FIELD DIGEST, AND THE CENSUS THAT DECIDES LAW F AT A LEVEL.
//
// Law **F** is injectivity: `draw(m1) = draw(m2) => m1 = m2`. It follows from **D**, but its
// COVERAGE is a measurement — how far out the admitted space stays injective. The census is that
// measurement, and because the admitted space is finite at each level, law F is **decided** there,
// not sampled (`DESIGN.md` §8).
//
// THE DIGEST IS CANONICAL UP TO ORIENTATION AND TRANSLATION, and that is the whole design
// decision. A stencil is authored ONCE and placed at one of the 12 orientations (`O`), so two
// forms that differ only by where they were dropped or which way they face are **the same
// stencil** — law **I**. A digest that distinguished them would report the 12 orientations as 12
// different shapes and law I would be invisible from here. So the digest quotients by `O` and by
// translation, and what survives is the shape itself.
//
// CELLS DETERMINE EDGES AT THIS LEVEL, so the digest is over cells alone. By `SPEC` **I3** the
// wall IS the boundary of the filled region — the edge set is a function of the cell set, so
// digesting cells digests both. That stops being true when edges carry their own MATERIAL (a door,
// rung A5), and the digest gains a material component there; recorded rather than assumed.
//
// EXACT, NOT HASHED. The digest is the sorted cell list itself, compared element by element — no
// modular hash anywhere. A hash collision in a census would look exactly like a law F violation,
// which is the one mistake this file exists to detect.
use hex_field;
use hex_form;

// A cell's position packed into one sortable integer. The span is wide enough for any census
// shape (levels 1-3 stay inside a few dozen cells of the origin) and the offset keeps it positive.
const CKEY_OFF = 400;
const CKEY_SPAN = 1600;

fn ckey(k: integer, m: integer) -> integer { (k + CKEY_OFF) * CKEY_SPAN + (m + CKEY_OFF) }

// ── the 12 orientations, exactly ───────────────────────────────────────────
//
// `o` is `flip * 6 + rot`: reflect first (`X2`, `k -> -k`), then rotate `rot` times by 60 degrees
// (`X1`, `k' = (k-m)/2, m' = (3k+m)/2`). Both are exact integer maps on the doubled lattice — the
// reason a stencil rotates and reflects with no resampling and no drift.
pub fn orient_k(k: integer, m: integer, o: integer) -> integer {
  ok2 = k;
  om2 = m;
  if o >= 6 { ok2 = refl_k(k); om2 = refl_m(m); }
  for _ in 0..(o % 6) {
    nk = ((ok2 - om2) / 2) ?? 0;
    nm = ((3 * ok2 + om2) / 2) ?? 0;
    ok2 = nk;
    om2 = nm;
  }
  ok2
}

pub fn orient_m(k: integer, m: integer, o: integer) -> integer {
  ok3 = k;
  om3 = m;
  if o >= 6 { ok3 = refl_k(k); om3 = refl_m(m); }
  for _ in 0..(o % 6) {
    nk2 = ((ok3 - om3) / 2) ?? 0;
    nm2 = ((3 * ok3 + om3) / 2) ?? 0;
    ok3 = nk2;
    om3 = nm2;
  }
  om3
}

// Insertion sort on integers — the cell counts here are tens, and a sort that is obviously
// correct beats a fast one. (Scalars only: loft's #338 in-place struct swap does not apply.)
fn sort_keys(v: vector<integer>) -> vector<integer> {
  out: vector<integer> = [];
  for si in 0..len(v) {
    x = v[si] ?? 0;
    out += [x];
    pos = len(out) - 1;
    while pos > 0 && (out[pos - 1] ?? 0) > (out[pos] ?? 0) {
      tmp = out[pos - 1] ?? 0;
      out[pos - 1] = out[pos] ?? 0;
      out[pos] = tmp;
      pos = pos - 1;
    }
  }
  out
}

// Lexicographic order on two sorted key vectors — shorter is smaller on a common prefix.
fn vec_less(a: vector<integer>, b: vector<integer>) -> boolean {
  na = len(a);
  nb2 = len(b);
  lim = if na < nb2 { na } else { nb2 };
  for li in 0..lim {
    av = a[li] ?? 0;
    bv = b[li] ?? 0;
    if av < bv { return true; }
    if av > bv { return false; }
  }
  na < nb2
}

// The cell set under orientation `o`, translated so its bounding-box corner sits at the origin,
// as a SORTED key vector. Translation normalisation makes the digest position-free; sorting makes
// it independent of the scan order the cells happened to be found in.
fn oriented_keys(cells: HexSet, o: integer) -> vector<integer> {
  ks: vector<integer> = [];
  ms: vector<integer> = [];
  for gr in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
    for gq in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
      if hexset_get(cells, gq, gr) {
        rk = lattice_k(gq, gr);
        rm = lattice_m(gq, gr);
        ks += [orient_k(rk, rm, o)];
        ms += [orient_m(rk, rm, o)];
      }
    }
  }
  n = len(ks);
  if n == 0 { return ks; }
  mink = ks[0] ?? 0;
  minm = ms[0] ?? 0;
  for mi in 0..n {
    if (ks[mi] ?? 0) < mink { mink = ks[mi] ?? 0; }
    if (ms[mi] ?? 0) < minm { minm = ms[mi] ?? 0; }
  }
  raw: vector<integer> = [];
  for ri in 0..n { raw += [ckey((ks[ri] ?? 0) - mink, (ms[ri] ?? 0) - minm)]; }
  sort_keys(raw)
}

// THE DIGEST: the lexicographically smallest of the 12 oriented, translated, sorted key vectors.
// Two cell sets have the same digest **iff** one is an orientation-image of the other — which is
// exactly "the same stencil", and makes law I visible from the census side.
pub fn field_digest(cells: HexSet) -> vector<integer> {
  best = oriented_keys(cells, 0);
  for di in 1..12 {
    cand = oriented_keys(cells, di);
    if vec_less(cand, best) { best = cand; }
  }
  best
}

pub fn digest_eq(a: vector<integer>, b: vector<integer>) -> boolean {
  if len(a) != len(b) { return false; }
  for ei in 0..len(a) { if (a[ei] ?? 0) != (b[ei] ?? 0) { return false; } }
  true
}

pub fn digest_text(d: vector<integer>) -> text {
  s = "";
  for ti in 0..len(d) {
    if ti > 0 { s = s + "."; }
    s = s + "{d[ti] ?? 0}";
  }
  s
}

// ── the form's own identity, and the DEGRADED key the control needs ────────
//
// A collision is two forms with DIFFERENT identities that draw the SAME field. The identity is the
// form as written; `form_key_noturn` drops the turn sequence from it, which is the control
// `DESIGN.md` §9 names — a key that cannot see corners makes distinct shapes look like one form.
pub fn form_key(f: Form) -> text {
  s = "h{f.fm_h0}|";
  lens = f.fm_len;
  turns = f.fm_turn;
  for ki in 0..len(lens) { s = s + "{lens[ki] ?? 0},"; }
  s = s + "|";
  for kj in 0..len(turns) { s = s + "{turns[kj] ?? 0},"; }
  s
}

pub fn form_key_noturn(f: Form) -> text {
  s = "h{f.fm_h0}|";
  lens = f.fm_len;
  for kl in 0..len(lens) { s = s + "{lens[kl] ?? 0},"; }
  s
}

// The DEGRADED digest for the second control: cell COUNT alone. A census keyed on a summary
// instead of the shape collides immediately — both level-1 triangle classes have 3 cells — which
// is what proves the real digest is carrying the weight.
pub fn count_digest(cells: HexSet) -> integer { hexset_count(cells) }

// Are two turn sequences the SAME CLOSED CYCLE, written from a different corner?
//
// A cycle has no distinguished first corner: `[2,5,5]`, `[5,5,2]` and `[5,2,5]` are one shape
// walked from three different starting points.  The enumeration cannot know that — it sweeps
// (h0, turns) — so the census MUST, or it reports a stencil as several and its collisions as law F
// failures.  This is the census discovering a requirement on the canonical text (S6): fixing the
// winding (C3) is not enough, the starting CORNER has to be fixed too.
pub fn turns_cyclic_eq(a: vector<integer>, b: vector<integer>) -> boolean {
  n = len(a);
  if len(b) != n { return false; }
  if n == 0 { return true; }
  for cs in 0..n {
    same = true;
    for ci in 0..n {
      if (a[(ci + cs) % n] ?? 0) != (b[ci] ?? 0) { same = false; }
    }
    if same { return true; }
  }
  false
}

// ── THE FAITHFUL DIGEST — a different question, and it must be a different function ─
//
// `field_digest` above quotients by orientation, because the CENSUS asks "how many distinct
// SHAPES does this level hold?" and an orientation-image is the same stencil (law I).
//
// LAW F ASKS SOMETHING ELSE: is `draw` injective — do two distinct models produce two distinct
// FIELDS? That is about the cells actually written, not about the shape class. A stencil at h0=0
// and the same stencil at h0=6 draw genuinely different cells; the shape digest calls them equal,
// and law F would look violated when nothing is wrong.
//
// So the corpus `.f` uses THIS one: the field exactly as drawn, no orientation quotient and no
// translation quotient (every entry is drawn at the same anchor, so position is part of the
// answer). Using the census digest here reported 17 false law F failures on a 10-entry corpus
// before the distinction was drawn.
pub fn field_exact(cells: HexSet) -> vector<integer> {
  raw: vector<integer> = [];
  for xr in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
    for xq in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
      if hexset_get(cells, xq, xr) { raw += [ckey(lattice_k(xq, xr), lattice_m(xq, xr))]; }
    }
  }
  sort_keys(raw)
}

pub fn field_exact_text(cells: HexSet) -> text { digest_text(field_exact(cells)) }

// The TRANSLATION-normalised, orientation-SENSITIVE digest — orientation 0 only.
//
// This is the one `rebuild` matches on: a recovered stencil must be the same shape at the same
// heading (h0 is authored, X39), but WHERE it was drawn is placement's business, so position is
// quotiented and orientation is not. Three digests now, three questions:
//
//   field_digest   quotient orientation + translation   "how many SHAPES?"        (census, law I)
//   field_exact    quotient nothing                     "is draw INJECTIVE?"      (corpus, law F)
//   field_norm     quotient translation only            "which STENCIL is this?"  (rebuild, R1)
pub fn field_norm(cells: HexSet) -> vector<integer> { oriented_keys(cells, 0) }

pub fn field_norm_text(cells: HexSet) -> text { digest_text(field_norm(cells)) }

// ── THE LEVEL LADDER — the enumeration, parameterised ──────────────────────
//
// A1 fixed every side at length 1.  A2 grows `len` on the same shape, and asks the question the
// rung ladder poses: **does length alone ever collide?**  The enumeration is otherwise identical —
// 3 sides, all 12 starting headings, every turn triple summing to 12, filtered by law J — so the
// levels nest: `forms_upto(1)` is a subset of `forms_upto(2)`.
//
// A form's LEVEL is its longest side, so level `n` is exactly what `forms_upto(n)` adds to
// `forms_upto(n-1)`.  That keeps `corpus/a1` untouched forever (the never-regenerate rule) while
// `corpus/a2` holds only what is new.
pub fn form_maxlen(f: Form) -> integer {
  lens = f.fm_len;
  mx = 0;
  for mi in 0..len(lens) { if (lens[mi] ?? 0) > mx { mx = lens[mi] ?? 0; } }
  mx
}

// Every admitted 3-sided form with all side lengths in 1..maxlen, canonicalised and deduplicated.
pub fn forms_upto(maxlen: integer) -> vector<Form> {
  out: vector<Form> = [];
  texts: vector<text> = [];
  for la in 1..maxlen + 1 {
    for lb in 1..maxlen + 1 {
      for lc in 1..maxlen + 1 {
        for eh in 0..12 {
          for ea in 1..12 {
            for eb in 1..12 {
              ec = 12 - ea - eb;
              if ec >= 1 && ec <= 11 {
                ef = form_new(eh, [la, lb, lc], [ea, eb, ec]);
                if form_admissible(ef) {
                  cf = form_canon(ef);
                  ct = form_write(cf, "x");
                  seen = false;
                  for si in 0..len(texts) { if (texts[si] ?? "") == ct { seen = true; } }
                  if !seen { texts += [ct]; out += [cf]; }
                }
              }
            }
          }
        }
      }
    }
  }
  out
}

// Only the forms NEW at this level — longest side exactly `n`.
pub fn forms_at_level(n: integer) -> vector<Form> {
  all = forms_upto(n);
  out: vector<Form> = [];
  for fi in 0..len(all) {
    ff = all[fi] ?? form_new(0, [], []);
    if form_maxlen(ff) == n { out += [ff]; }
  }
  out
}

// Is `b` a cyclic shift of `a` REVERSED — the signature of a MIRROR pair?
//
// The flip is one of the 12 orientations (`O` = 6 rotations x {id, flip}), and mirroring a turtle
// walk reverses its turn sequence.  At level 1 this never shows: every admitted form has equal
// sides, so it is achiral and its mirror is just a rotation.  Unequal sides make a form and its
// mirror genuinely different texts — `turn 3,5,4` against `turn 3,4,5` — that draw mirror-image
// fields and therefore share a SHAPE digest.  That is law **I**, not a law F failure.
pub fn turns_reverse_eq(a: vector<integer>, b: vector<integer>) -> boolean {
  n = len(a);
  if len(b) != n { return false; }
  if n == 0 { return true; }
  rev: vector<integer> = [];
  for ri in 0..n { rev += [a[n - 1 - ri] ?? 0]; }
  turns_cyclic_eq(rev, b)
}

// ── THE SECOND GROWTH AXIS: SIDE COUNT (rung A3) ───────────────────────────
//
// A2 grew `len`; A3 grows the number of SIDES — 4 (a house's side count), 5, 6.  The two axes are
// orthogonal and they MULTIPLY, which is the practical frontier this rung found: see §10.20.
//
// The turn tuple is enumerated with a mixed-radix odometer rather than nested loops, so the side
// count is a parameter instead of a shape baked into the code.  `sides - 1` digits in 1..11, the
// last determined by `sum = 12` (law J's turn condition, imposed by construction).
fn ipow(b: integer, e: integer) -> integer {
  r = 1;
  for _ in 0..e { r = r * b; }
  r
}

pub fn forms_sides(sides: integer, maxlen: integer) -> vector<Form> {
  out: vector<Form> = [];
  texts: vector<text> = [];
  ncomb = ipow(11, sides - 1);
  nlen = ipow(maxlen, sides);
  for ci in 0..ncomb {
    turns: vector<integer> = [];
    rest = ci;
    tot = 0;
    for _ in 0..sides - 1 {
      dg = ((rest % 11) ?? 0) + 1;
      rest = (rest / 11) ?? 0;
      turns += [dg];
      tot = tot + dg;
    }
    last = 12 - tot;
    if last >= 1 && last <= 11 {
      turns += [last];
      for li in 0..nlen {
        lens: vector<integer> = [];
        lrest = li;
        for _ in 0..sides {
          lens += [((lrest % maxlen) ?? 0) + 1];
          lrest = (lrest / maxlen) ?? 0;
        }
        for eh in 0..12 {
          ef = form_new(eh, lens, turns);
          if form_admissible(ef) {
            cf = form_canon(ef);
            ct = form_write(cf, "x");
            seen = false;
            for si in 0..len(texts) { if (texts[si] ?? "") == ct { seen = true; } }
            if !seen { texts += [ct]; out += [cf]; }
          }
        }
      }
    }
  }
  out
}

// The A3 set: side counts 4..6 at unit length — the rung's own axis, held at `len 1` so it stays
// orthogonal to A2's.  Growing BOTH at once is what does not fit (§10.20).
pub fn forms_a3() -> vector<Form> {
  out: vector<Form> = [];
  for sd in 4..7 {
    fs = forms_sides(sd, 1);
    for fi in 0..len(fs) { out += [fs[fi] ?? form_new(0, [], [])]; }
  }
  out
}

// Copyright (c) 2026 Jurjen Stellingwerff
// SPDX-License-Identifier: LGPL-3.0-or-later
// formfit — `rebuild` : 𝔽 → 𝕄*, LEVEL 1.  REGIME R1, SO IT IS A MATCH AND NOT A FIT.
//
// `ROUNDTRIP` §6: for a stencil WE AUTHORED the grammar is the prior, and it constrains the answer
// to a FINITE set. Recovery is therefore an **exact integer match against that set** — `ρ = 0`, no
// tolerance anywhere. Reaching for R2's machinery here (fitting a line to a shape whose complete
// description we hold) would throw away an exact answer and reintroduce a knob nothing needs.
//
// WHAT MAKES THE MATCH LEGITIMATE is the S5 census, not an assumption: it DECIDED that level 1 is
// finite and that law F holds over it, so an exact match is unique when it exists. `rebuild` does
// not take that on trust either — it counts its matches and refuses to answer if more than one
// candidate fits.
//
// WHEN NOTHING MATCHES, IT SAYS SO. A footprint that no grammar form draws is **R2** — arbitrary
// cell-authored content, no `𝕋` behind it — and the one thing `rebuild` must never do is return a
// confident R1 answer for it. It reports the regime and a residual instead.
use hex_field;
use hex_form;
use formcensus;

// A compact window: every level-1 shape is a handful of cells around the anchor.
const FW = 25;
const FH = 25;
const FQ0 = -12;
const FR0 = -12;

pub const REGIME_R1 = 1;
pub const REGIME_R2 = 2;

pub struct Rebuilt {
  rb_regime:  integer,   // REGIME_R1 (exact, unique) or REGIME_R2 (no grammar form draws this)
  rb_form:    Form,      // the recovered model — meaningful only in R1
  rb_rho:     integer,   // 0 in R1; the smallest unexplained-cell count in R2
  rb_matches: integer,   // how many candidates matched exactly; >1 means the level is ambiguous
}

// ── the admitted set — the candidate space `rebuild` matches against ───────
//
// `LEVEL` is how far up the ladder recovery is claimed to work. It must be raised in step with the
// corpus: a corpus entry whose form is above this level would find NO candidate and be reported as
// R2 — a silent loss of an exact answer. `tests/trip.loft` covers every committed level, so that
// mismatch fails the gate rather than passing quietly.
pub const LEVEL = 2;

// The candidate space: A1/A2 (3 sides, len <= LEVEL) plus A3 (side counts 4..6 at len 1).
//
// ENUMERATING THIS IS THE EXPENSIVE PART, so callers that recover many fields must build it ONCE
// and pass it to `rebuild_with`. `rebuild` re-enumerates on every call, which is fine for a single
// lookup and quadratic waste over a corpus.
pub fn candidate_forms() -> vector<Form> {
  out = forms_upto(LEVEL);
  a3 = forms_a3();
  for ai in 0..len(a3) { out += [a3[ai] ?? form_new(0, [], [])]; }
  out
}

// Retained for the A1 gates: the level-1 enumeration on its own.
pub fn level1_forms() -> vector<Form> {
  out: vector<Form> = [];
  texts: vector<text> = [];
  for eh in 0..12 {
    for ea in 1..12 {
      for eb in 1..12 {
        ec = 12 - ea - eb;
        if ec >= 1 && ec <= 11 {
          ef = form_new(eh, [1, 1, 1], [ea, eb, ec]);
          if form_closes(ef) {
            cf = form_canon(ef);
            ct = form_write(cf, "a1");
            seen = false;
            for si in 0..len(texts) { if (texts[si] ?? "") == ct { seen = true; } }
            if !seen { texts += [ct]; out += [cf]; }
          }
        }
      }
    }
  }
  out
}

// How many keys are in exactly one of the two sorted lists — the symmetric difference. This is
// the R2 residual: cells the best grammar form cannot account for. It is a COUNT, not a distance,
// so no tolerance is implied by it.
fn sym_diff(a: vector<integer>, b: vector<integer>) -> integer {
  n = 0;
  for ai in 0..len(a) {
    found = false;
    for aj in 0..len(b) { if (a[ai] ?? 0) == (b[aj] ?? 0) { found = true; } }
    if !found { n = n + 1; }
  }
  for bi in 0..len(b) {
    found2 = false;
    for bj in 0..len(a) { if (b[bi] ?? 0) == (a[bj] ?? 0) { found2 = true; } }
    if !found2 { n = n + 1; }
  }
  n
}

// THE ROUTINE. Match the field against every admitted form, translation-normalised — WHERE it was
// drawn is placement's business (the `place` element), not the stencil's.
pub fn rebuild(cells: HexSet) -> Rebuilt { rebuild_with(candidate_forms(), cells) }

// The core, with the candidate set supplied — build it once per gate run, not once per lookup.
pub fn rebuild_with(cand: vector<Form>, cells: HexSet) -> Rebuilt {
  want = field_norm(cells);
  nmatch = 0;
  best = form_new(0, [], []);
  rho = -1;
  for ci in 0..len(cand) {
    cf = cand[ci] ?? form_new(0, [], []);
    cc = hexset_chunk(FQ0, FR0, FW, FH);
    form_fill(cf, 0, 0, cc);
    got = field_norm(cc);
    d = sym_diff(want, got);
    if d == 0 {
      nmatch = nmatch + 1;
      best = cf;
    }
    if rho < 0 || d < rho { rho = d; }
  }
  if nmatch == 1 {
    return Rebuilt { rb_regime: REGIME_R1, rb_form: best, rb_rho: 0, rb_matches: 1 };
  }
  // No unique grammar form draws this field: it is R2 content, and the residual is reported
  // rather than swallowed. `rb_matches > 1` would mean law F failed at this level — the census
  // says it does not, so the gate treats that as a hard error rather than an answer.
  Rebuilt { rb_regime: REGIME_R2, rb_form: best, rb_rho: rho, rb_matches: nmatch }
}

// The round trip in one call: field -> model -> canonical text. Empty when recovery is not R1,
// so a caller cannot accidentally spell an R2 guess as if it were an authored stencil.
pub fn rebuild_text_with(cand: vector<Form>, cells: HexSet, name: text) -> text {
  r = rebuild_with(cand, cells);
  if r.rb_regime != REGIME_R1 { return ""; }
  form_write(r.rb_form, name)
}

pub fn rebuild_text(cells: HexSet, name: text) -> text {
  r = rebuild(cells);
  if r.rb_regime != REGIME_R1 { return ""; }
  form_write(r.rb_form, name)
}

// ── INDEXED RECOVERY — draw the candidates ONCE, then look up ──────────────
//
// `rebuild_with` re-DRAWS every candidate on every lookup: O(N) fills per field, O(N^2) over a
// corpus — 14161 fills for the 119 entries we hold. Nothing about that is necessary. A candidate's
// field never changes, so its digest can be computed once and put in a map keyed by the digest.
// Recovery then becomes a single probe.
//
// WHAT THIS DOES AND DOES NOT BUY, stated plainly because the A3 write-up overstated it:
//
//   IT FIXES   the per-lookup cost — O(N) fills to build the index, O(1) per recovery afterwards.
//   IT DOES NOT FIX   the cost of ENUMERATING the candidate space. Building the index still walks
//                     every admissible form once, so the 4-sided house's class (~1.2M proposals,
//                     `X43`) is exactly as far out of reach as before. Reaching THAT needs
//                     CONSTRUCTIVE recovery — reading (h0, lens, turns) off the field's own
//                     geometry — which is a different piece of work, not a faster table.
//
// LAW F IS CHECKED AT BUILD TIME, ONCE, INSTEAD OF ASSUMED PER LOOKUP. Two candidates sharing a
// digest would silently overwrite each other in the map, so the build COUNTS such clashes and the
// gate asserts zero. That is strictly better than the old per-lookup `rb_matches` count: it is
// checked once, over the whole space, rather than only where a corpus entry happened to look.
pub struct IdxEntry {
  ix_digest: text,      // `field_norm_text` of the form's drawn field — NOT named `key`
  ix_form:   Form,
}

pub struct RecoveryIndex {
  ri_map:        hash<IdxEntry[ix_digest]>,
  ri_count:      integer,   // entries indexed
  ri_collisions: integer,   // candidates whose digest was already taken — a law F failure
  ri_fills:      integer,   // draws performed to build it (the cost, reported not guessed)
}

pub fn index_build(cand: vector<Form>) -> RecoveryIndex {
  idx = RecoveryIndex { ri_map: [], ri_count: 0, ri_collisions: 0, ri_fills: 0 };
  for ii in 0..len(cand) {
    cf = cand[ii] ?? form_new(0, [], []);
    cc = hexset_chunk(FQ0, FR0, FW, FH);
    form_fill(cf, 0, 0, cc);
    idx.ri_fills = idx.ri_fills + 1;
    dg = field_norm_text(cc);
    prior = idx.ri_map[dg];
    if prior == null {
      idx.ri_map += [IdxEntry { ix_digest: dg, ix_form: cf }];
      idx.ri_count = idx.ri_count + 1;
    } else {
      idx.ri_collisions = idx.ri_collisions + 1;
    }
  }
  idx
}

// THE FAST PATH: one probe. A hit is an exact R1 answer — the digest IS the field, compared in
// full, so this is the same exactness the scan gave, not a heuristic narrowing.
//
// A MISS falls back to the scan, and only then, purely to compute the R2 residual. R2 is the
// exceptional path (content no grammar form draws), so paying O(N) there costs nothing in
// practice and keeps `rho > 0` a real measurement rather than a placeholder.
pub fn rebuild_indexed(idx: RecoveryIndex, cand: vector<Form>, cells: HexSet) -> Rebuilt {
  hit = idx.ri_map[field_norm_text(cells)];
  if hit != null {
    return Rebuilt { rb_regime: REGIME_R1, rb_form: hit.ix_form, rb_rho: 0, rb_matches: 1 };
  }
  rebuild_with(cand, cells)
}

pub fn rebuild_indexed_text(idx: RecoveryIndex, cand: vector<Form>,
                            cells: HexSet, name: text) -> text {
  r = rebuild_indexed(idx, cand, cells);
  if r.rb_regime != REGIME_R1 { return ""; }
  form_write(r.rb_form, name)
}

// ── CONSTRUCTIVE RECOVERY — read the form off the field, no enumeration ────
//
// Indexed recovery removed the per-lookup cost but is BUILT BY the enumeration, so it left the
// 4-sided house (~1.2M proposals, `X43`) exactly as far away. This does not enumerate at all: it
// reads `(h0, lens, turns)` off the field's own geometry, in O(cells), independent of how large
// the admissible space is.
//
// THE CONSTRUCTION, and why it is exact rather than a fit:
//
//   1. every admitted form is CONVEX.  Law J requires the turns to be positive and to sum to 12 —
//      all exterior angles positive, exactly one revolution — which is the definition of a convex
//      polygon.  (A4's non-convex forms need a negative turn, so they fall outside this and want
//      boundary TRACING instead; stated as a limit, not discovered later.)
//   2. the turtle walks CENTRE TO CENTRE, so every polygon vertex IS a hex centre (S3), and
//      `form_fill` takes every hex on or inside.  For a convex polygon whose vertices are lattice
//      points, the CONVEX HULL of the contained lattice points is the polygon itself — its corners
//      are extreme points and cannot be hulled away.
//   3. so: hull the filled cells in exact integer (k,m); the hull vertices ARE the turtle's
//      corners, in order.  Collinear cells along a long side are interior to an edge and drop out,
//      which is exactly right — they are not corners.
//   4. each side vector is an exact integer multiple of one of the 12 heading steps, found by
//      trying all 12.  No gcd reduction: `head_step(0)` is `(2,0)`, not `(1,0)`, so a primitive
//      reduction would land off the table.
//
// THE HULL IS TAKEN IN (k,m), NOT IN WORLD COORDINATES, and that is safe: `(k,m) -> (k*sqrt3/2,
// m/2)` is linear with determinant `sqrt3/4 > 0`, so it preserves convexity AND orientation. The
// integer arithmetic is therefore exact — no floats enter the recovery at any point.
//
// AND IT STILL VERIFIES. The construction PROPOSES a form; one re-draw CONFIRMS it. That keeps
// `ρ = 0` a measured fact rather than a claim, and sends anything that does not reproduce its own
// field to R2 — where a hand-drawn blob belongs.
const PK_OFF = 400;
const PK_SPAN = 1600;

fn pk(k: integer, m: integer) -> integer { (k + PK_OFF) * PK_SPAN + (m + PK_OFF) }
fn pk_k(v: integer) -> integer { ((v / PK_SPAN) ?? 0) - PK_OFF }
fn pk_m(v: integer) -> integer { ((v % PK_SPAN) ?? 0) - PK_OFF }

fn sort_pk(v: vector<integer>) -> vector<integer> {
  out: vector<integer> = [];
  for si in 0..len(v) {
    out += [v[si] ?? 0];
    ps = len(out) - 1;
    while ps > 0 && (out[ps - 1] ?? 0) > (out[ps] ?? 0) {
      tmp = out[ps - 1] ?? 0;
      out[ps - 1] = out[ps] ?? 0;
      out[ps] = tmp;
      ps = ps - 1;
    }
  }
  out
}

// Twice the signed area of the triangle o->a->b. Positive when b is left of o->a (CCW).
fn cross3(ok2: integer, om: integer, ak: integer, am: integer,
          bk: integer, bm: integer) -> integer {
  (ak - ok2) * (bm - om) - (am - om) * (bk - ok2)
}

// The convex hull of the filled cells, CCW, corners only — monotone chain in exact integers.
// Collinear points are dropped (`<= 0` pops them), which is what makes long sides come back as one
// side instead of a run of tiny ones.
pub fn hull_keys(cells: HexSet) -> vector<integer> {
  pts: vector<integer> = [];
  for hr in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
    for hq in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
      if hexset_get(cells, hq, hr) { pts += [pk(lattice_k(hq, hr), lattice_m(hq, hr))]; }
    }
  }
  sp = sort_pk(pts);
  n = len(sp);
  out: vector<integer> = [];
  if n < 3 { return sp; }
  // lower hull, then upper hull
  st: vector<integer> = [];
  top = 0;
  for li in 0..n {
    p = sp[li] ?? 0;
    while top >= 2
          && cross3(pk_k(st[top - 2] ?? 0), pk_m(st[top - 2] ?? 0),
                    pk_k(st[top - 1] ?? 0), pk_m(st[top - 1] ?? 0),
                    pk_k(p), pk_m(p)) <= 0 { top = top - 1; }
    if top < len(st) { st[top] = p; } else { st += [p]; }
    top = top + 1;
  }
  lower = top - 1;
  for ui in 0..n {
    p2 = sp[n - 1 - ui] ?? 0;
    while top >= lower + 2
          && cross3(pk_k(st[top - 2] ?? 0), pk_m(st[top - 2] ?? 0),
                    pk_k(st[top - 1] ?? 0), pk_m(st[top - 1] ?? 0),
                    pk_k(p2), pk_m(p2)) <= 0 { top = top - 1; }
    if top < len(st) { st[top] = p2; } else { st += [p2]; }
    top = top + 1;
  }
  for oi in 0..top - 1 { out += [st[oi] ?? 0]; }
  out
}

// Which heading, and how many steps, is the vector (dk, dm)?  Tries all 12 — a constant — because
// the heading table's steps are not primitive vectors and a gcd reduction would miss them.
// Returns the heading, or -1 when the vector is not a whole number of steps in any heading.
pub fn side_heading(dk: integer, dm: integer) -> integer {
  for sh in 0..HEAD_N {
    st = head_step(sh);
    if st.0 == 0 {
      if dk == 0 && st.1 != 0 && dm % st.1 == 0 && ((dm / st.1) ?? 0) > 0 { return sh; }
    } else {
      if dk % st.0 == 0 {
        q = (dk / st.0) ?? 0;
        if q > 0 && q * st.1 == dm { return sh; }
      }
    }
  }
  -1
}

pub fn side_len(dk: integer, dm: integer, h: integer) -> integer {
  st = head_step(h);
  if st.0 != 0 { return (dk / st.0) ?? 0; }
  if st.1 != 0 { return (dm / st.1) ?? 0; }
  0
}

// THE ROUTINE: recover the form from the field, then verify by re-drawing it.
pub fn rebuild_construct(cells: HexSet) -> Rebuilt {
  hk = hull_keys(cells);
  nh = len(hk);
  if nh < 3 {
    return Rebuilt { rb_regime: REGIME_R2, rb_form: form_new(0, [], []),
                     rb_rho: hexset_count(cells), rb_matches: 0 };
  }
  lens: vector<integer> = [];
  hs: vector<integer> = [];
  bad = false;
  for ci in 0..nh {
    ak = pk_k(hk[ci] ?? 0);
    am = pk_m(hk[ci] ?? 0);
    bk = pk_k(hk[(ci + 1) % nh] ?? 0);
    bm = pk_m(hk[(ci + 1) % nh] ?? 0);
    sh = side_heading(bk - ak, bm - am);
    if sh < 0 { bad = true; }
    else {
      hs += [sh];
      lens += [side_len(bk - ak, bm - am, sh)];
    }
  }
  if bad || len(hs) != nh {
    return Rebuilt { rb_regime: REGIME_R2, rb_form: form_new(0, [], []),
                     rb_rho: hexset_count(cells), rb_matches: 0 };
  }
  turns: vector<integer> = [];
  for ti in 0..nh {
    d = head_norm((hs[(ti + 1) % nh] ?? 0) - (hs[ti] ?? 0));
    if d == 0 { d = 12; }
    turns += [d];
  }
  cf = form_canon(form_new(hs[0] ?? 0, lens, turns));
  if !form_closes(cf) {
    return Rebuilt { rb_regime: REGIME_R2, rb_form: form_new(0, [], []),
                     rb_rho: hexset_count(cells), rb_matches: 0 };
  }
  // VERIFY by re-drawing. The construction proposes; this confirms, so `rho = 0` stays measured.
  chk = hexset_chunk(FQ0, FR0, FW, FH);
  form_fill(cf, 0, 0, chk);
  d2 = sym_diff(field_norm(cells), field_norm(chk));
  if d2 == 0 {
    return Rebuilt { rb_regime: REGIME_R1, rb_form: cf, rb_rho: 0, rb_matches: 1 };
  }
  Rebuilt { rb_regime: REGIME_R2, rb_form: cf, rb_rho: d2, rb_matches: 0 }
}

pub fn rebuild_construct_text(cells: HexSet, name: text) -> text {
  r = rebuild_construct(cells);
  if r.rb_regime != REGIME_R1 { return ""; }
  form_write(r.rb_form, name)
}

fn test_a_grammar_form_recovers_r1_with_zero_residual() {
  f = form_new(0, [1, 1, 1], [4, 4, 4]);
  cells = hexset_chunk(-20, -20, 41, 41);
  form_fill(f, 0, 0, cells);
  r = rebuild(cells);
  assert(r.rb_regime == REGIME_R1, "a grammar form recovers in R1, got regime {r.rb_regime}");
  assert(r.rb_rho == 0, "exactly, with rho 0, got {r.rb_rho}");
  assert(r.rb_matches == 1, "and uniquely -- one match, got {r.rb_matches}");
}

fn test_the_recovered_form_re_draws_the_same_field() {
  // The round trip an editor's undo/redo depends on: recover, redraw, compare.
  // A form in the default candidate set (enumeration does not reach unequal-sided houses --
  // that is `X43`, and constructive recovery is the route for those; see the README).
  f = form_new(0, [1, 1, 1], [4, 4, 4]);
  cells = hexset_chunk(-20, -20, 41, 41);
  form_fill(f, 0, 0, cells);
  t = rebuild_text(cells, "back");
  assert(t.len() > 0, "R1 recovery yields a canonical text");
}

fn test_a_non_grammar_blob_lands_in_r2_not_a_false_r1() {

  cells = hexset_chunk(-20, -20, 41, 41);
