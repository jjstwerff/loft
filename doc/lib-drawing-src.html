// Copyright (c) 2026 Jurjen Stellingwerff
// SPDX-License-Identifier: LGPL-3.0-or-later
//
// drawing — a sprite authored as lines of text, rendered in loft.
//
// A `.draw` scene is a dozen lines of marks on fractional paper:
//
//     size 256x256
//     Background transparent
//     name blade
//     Poly (0.48,0.10) (0.52,0.10) (0.52,0.70) (0.48,0.70) rgb=190,196,206
//     Circle (0.50,0.78) r=0.06 rgb=120,90,50
//
// ── ⚠⚠ THE INVARIANT ────────────────────────────────────────────────
//
// > **The picture is the one `draw.py` renders — pixel for pixel, not
// > approximately.**
//
// That is not a quality bar, it is the definition.  A corpus of `.draw` scenes
// already exists and already has art in it; the Python renderer is what made
// that art look the way it does, so "close enough" here means every sprite in
// the corpus quietly changes the day it is re-rendered.  The whole port is
// therefore measured by a byte diff against Pillow's output and nothing else.
//
// Three things follow from that, and each of them cost a probe to learn:
//
//  1. **The rasteriser is Pillow's, not `graphics`'.**  `graphics::fill_polygon`
//     is a fine polygon filler that obeys a different rule, and the two agree on
//     4 of 400 random polygons.  See [raster.loft](raster.loft).
//  2. **The picture is drawn at 3x and resampled with Lanczos.**  The
//     supersample factor and the filter are part of the contract a scene signs,
//     not an implementation choice — `graphics::resize_lanczos` is byte-identical
//     to `Image.resize(…, LANCZOS)` for exactly this reason.
//  3. **Absence is reported, never guessed.**  A line no command accepts is
//     collected and named rather than dropped, because a typo'd mark has to read
//     as a syntax problem and not as a geometry one.
//
// ── WHAT THIS BUILD DRAWS ───────────────────────────────────────────
//
// `size`, `Background`, `name`, `Line`, `Circle`, `Poly`, with solid fills
// (`rgb=` / `fill=`), coloured strokes (`stroke=`), pen widths (`w=`, and the
// per-point `@N` taper), and smooth points (`~`).  `grad=` / `radial=` fills and
// the `Petals` / `Fronds` arrays parse and are listed in `Sketch.deferred` — they
// are recognised, and a caller can see they were not drawn.
use graphics;
use raster;
use scan;

/// The supersample factor the corpus is drawn at.  Not adjustable: every golden
/// in the corpus carries it, and a scene rendered at any other factor is a
/// different picture.
pub const SUPERSAMPLE = 3;

/// The ink a mark uses when it names no colour.
pub const DEFAULT_INK_R = 38;
pub const DEFAULT_INK_G = 32;
pub const DEFAULT_INK_B = 36;

/// Control points per curve segment when a smooth point (`~`) expands.
const SMOOTH_SAMPLES = 10;

const PI = 3.141592653589793;

/// How a mark is filled, if it is filled at all.
pub enum PaintKind { Stroked, Solid, Linear, Radial }

/// A mark's fill descriptor.
///
/// `Stroked` is the absence of a fill, not a fill of nothing: it is what sends a
/// `Poly` down the pen path instead of the polygon path, and the two produce
/// completely different pictures from the same points.
pub struct Paint {
  pk: PaintKind = Stroked,
  c1: integer = 0,
  c2: integer = 0,
  /// `Linear`: `ax,ay,bx,by`.  `Radial`: `cx,cy,r`.  Empty means "derive it from
  /// the shape's own bounding box", which is what a fill with no `dir=` / `at=`
  /// asks for.
  spec: vector<float>,
}

/// What an op does.
pub enum OpKind { Sky, Fill, Stroke }

/// One entry in the ordered draw program.
///
/// ⚠ `pts` are FRACTIONS of the paper (0..1, origin top-left, y down), never
/// pixels.  The scale to pixels happens once, at the raster call, because the
/// paper size is a property of the scene and the supersample factor is a
/// property of the renderer — neither belongs in a parsed mark.
pub struct Op {
  kind: OpKind = Sky,
  pts: vector<raster::Pt>,
  paint: Paint,
  /// Stroke: one width per point (a taper).  Empty means the uniform `w`.
  widths: vector<float>,
  w: integer = 3,
  /// Stroke ink, or `Sky`'s top colour.
  color: integer = 0,
  /// `Sky`'s bottom colour.
  color2: integer = 0,
}

/// A named group of marks and the box they cover, in paper fractions.
pub struct Elem {
  ename: text = "",
  /// False until a mark has actually contributed a point: an element that is
  /// named and never drawn has no box, and answering `0,0,0,0` for it would be a
  /// measurement rather than an absence.
  seen: boolean = false,
  bx0: float = 0.0,
  by0: float = 0.0,
  bx1: float = 0.0,
  by1: float = 0.0,
}

/// A `landmark <name> = <value>` line.
pub struct Landmark {
  lname: text = "",
  lvalue: float = 0.0,
}

/// A parsed scene: the paper, the ordered draw program, and everything the
/// parser could measure or could not accept.
///
/// ⚠ NOT called `Scene`: `mesh3d::Scene` owns that name and `graphics` depends on
/// mesh3d, so the collision is a hard error — the same trap `graphics::Coord`
/// hit with `Point`.  Reach for `Sketch` in `.draw` code and the 3-D names stay
/// where they belong.
pub struct Sketch {
  sw: integer = 800,
  sh: integer = 800,
  transparent: boolean = false,
  ops: vector<Op>,
  elems: vector<Elem>,
  landmarks: vector<Landmark>,
  checks: vector<text>,
  /// `line N: <text>` for every line no command accepted.
  unparsed: vector<text>,
  /// `line N: <what>` for every mark this build recognises but does not draw.
  deferred: vector<text>,
}

// ── Colour ──────────────────────────────────────────────────────────

/// A grey level in 0..1 as an opaque colour, rounded HALF TO EVEN.
///
/// Python's `round` is half-to-even and `draw.py` uses it here, so `fill=0.5`
/// is 128 and not 127 — a difference of one level that no eye would find and a
/// byte diff would.
pub fn grey(gl: float) -> integer {
  gy = round_half_even(gl * 255.0) as integer;
  if gy < 0 { return graphics::rgba(0, 0, 0, 255) }
  if gy > 255 { return graphics::rgba(255, 255, 255, 255) }
  graphics::rgba(gy, gy, gy, 255)
}

fn round_half_even(rv: float) -> float {
  rh_f = floor(rv);
  rh_d = rv - rh_f;
  if rh_d > 0.5 { return rh_f + 1.0 }
  if rh_d < 0.5 { return rh_f }
  if ((rh_f as integer) % 2 ?? 0) == 0 { rh_f } else { rh_f + 1.0 }
}

fn default_ink() -> integer {
  graphics::rgba(DEFAULT_INK_R, DEFAULT_INK_G, DEFAULT_INK_B, 255)
}

// ── Option readers ──────────────────────────────────────────────────

/// `\bw\s*=\s*(\d+)` — the uniform pen width, or `fallback`.
fn read_width(s: text, fallback: integer) -> integer {
  rw_o = scan::find_option(s, "w");
  if !rw_o.ok { return fallback }
  rw_n = scan::read_uint(s, rw_o.at);
  if !rw_n.ok { return fallback }
  rw_n.value as integer
}

/// `\(?(\d+)\s*,\s*(\d+)\s*,\s*(\d+)\)?` at `i` — a colour triple, optionally
/// parenthesised, as `draw.py` writes every colour option.
fn read_rgb_at(s: text, i: integer) -> Scan3 {
  rr_i = i;
  rr_paren = scan::at(s, rr_i) == 40;
  if rr_paren { rr_i = scan::skip_space(s, rr_i + 1); }
  rr_r = scan::read_uint(s, rr_i);
  if !rr_r.ok { return scan3_fail() }
  rr_i = scan::skip_space(s, rr_r.at);
  if scan::at(s, rr_i) != 44 { return scan3_fail() }
  rr_g = scan::read_uint(s, scan::skip_space(s, rr_i + 1));
  if !rr_g.ok { return scan3_fail() }
  rr_i = scan::skip_space(s, rr_g.at);
  if scan::at(s, rr_i) != 44 { return scan3_fail() }
  rr_b = scan::read_uint(s, scan::skip_space(s, rr_i + 1));
  if !rr_b.ok { return scan3_fail() }
  rr_i = rr_b.at;
  if scan::at(s, rr_i) == 41 { rr_i += 1; }
  Scan3 { ok: true, at: rr_i,
          color: graphics::rgba(rr_r.value as integer, rr_g.value as integer,
                                rr_b.value as integer, 255) }
}

/// A colour read and where it ended.
struct Scan3 {
  ok: boolean = false,
  at: integer = 0,
  color: integer = 0,
}

fn scan3_fail() -> Scan3 { Scan3 { ok: false, at: 0, color: 0 } }

/// `\bstroke\s*=\s*R,G,B` — a pen colour.  `rgb=` still means FILL; this lets a
/// stroke carry colour without becoming one.
fn read_stroke_colour(s: text) -> integer {
  rs_o = scan::find_option(s, "stroke");
  if !rs_o.ok { return default_ink() }
  rs_c = read_rgb_at(s, rs_o.at);
  if !rs_c.ok { return default_ink() }
  rs_c.color
}

/// `draw.py`'s `_paint`, in its order: `radial=` beats `grad=` beats `rgb=`
/// beats `fill=`, and nothing at all means the mark is stroked.
fn read_paint(s: text) -> Paint {
  rp_rad = scan::find_option(s, "radial");
  if rp_rad.ok {
    rp_pair = read_colour_pair(s, rp_rad.at);
    if rp_pair.ok {
      return Paint { pk: Radial, c1: rp_pair.c1, c2: rp_pair.c2,
                     spec: read_spec(s, "at", 3) }
    }
  }
  rp_grad = scan::find_option(s, "grad");
  if rp_grad.ok {
    rp_pair2 = read_colour_pair(s, rp_grad.at);
    if rp_pair2.ok {
      return Paint { pk: Linear, c1: rp_pair2.c1, c2: rp_pair2.c2,
                     spec: read_spec(s, "dir", 4) }
    }
  }
  rp_rgb = scan::find_option(s, "rgb");
  if rp_rgb.ok {
    rp_c = read_rgb_at(s, rp_rgb.at);
    if rp_c.ok { return Paint { pk: Solid, c1: rp_c.color, c2: 0, spec: [] } }
  }
  rp_fill = scan::find_option(s, "fill");
  if rp_fill.ok {
    rp_l = scan::read_unsigned(s, rp_fill.at);
    if rp_l.ok { return Paint { pk: Solid, c1: grey(rp_l.value), c2: 0, spec: [] } }
  }
  Paint { pk: Stroked, c1: 0, c2: 0, spec: [] }
}

/// Two colours either side of a `>`.
struct ColourPair {
  ok: boolean = false,
  c1: integer = 0,
  c2: integer = 0,
}

fn read_colour_pair(s: text, i: integer) -> ColourPair {
  cp_a = read_rgb_at(s, i);
  if !cp_a.ok { return ColourPair { ok: false, c1: 0, c2: 0 } }
  cp_j = scan::skip_space(s, cp_a.at);
  if scan::at(s, cp_j) != 62 { return ColourPair { ok: false, c1: 0, c2: 0 } }
  cp_b = read_rgb_at(s, scan::skip_space(s, cp_j + 1));
  if !cp_b.ok { return ColourPair { ok: false, c1: 0, c2: 0 } }
  ColourPair { ok: true, c1: cp_a.color, c2: cp_b.color }
}

/// `\b<key>\s*=\s*n,n[,n[,n]]` — a comma-separated run of exactly `count`
/// numbers, or empty when the option is absent or short.
fn read_spec(s: text, key: text, count: integer) -> vector<float> {
  rs_out: vector<float> = [];
  rs_o = scan::find_option(s, key);
  if !rs_o.ok { return rs_out }
  rs_i = rs_o.at;
  for k in 0..count {
    if k > 0 {
      if scan::at(s, rs_i) != 44 { rs_out = []; return rs_out }
      rs_i += 1;
    }
    rs_n = scan::read_number(s, rs_i);
    if !rs_n.ok { rs_out = []; return rs_out }
    rs_out += [rs_n.value];
    rs_i = rs_n.at;
  }
  rs_out
}

// ── Points on a line ────────────────────────────────────────────────

/// Every `(x,y)` on a line, with the two suffixes a point may carry.
pub struct PointList {
  pts: vector<raster::Pt>,
  /// `~` — this point curves; a point without it is a corner.
  smooth: vector<boolean>,
  /// `@N` — the pen width AT this point.  `has_width` says which were given.
  widths: vector<float>,
  has_width: vector<boolean>,
  any_width: boolean = false,
}

/// `\(\s*(n)\s*,\s*(n)\s*\)\s*(~?)\s*(?:@\s*(n))?`, found repeatedly along the line.
///
/// Scanning for points rather than splitting the line is what lets the same
/// grammar carry options: `rgb=140,104,62` has no parentheses, so it is not a
/// point, and no separator has to be agreed between the two.
pub fn read_points(s: text) -> PointList {
  rp_out = PointList { pts: [], smooth: [], widths: [], has_width: [], any_width: false };
  rp_i = 0;
  rp_n = size(s);
  while rp_i < rp_n {
    if scan::at(s, rp_i) != 40 { rp_i += 1; continue }
    rp_j = scan::skip_space(s, rp_i + 1);
    rp_x = scan::read_number(s, rp_j);
    if !rp_x.ok { rp_i += 1; continue }
    rp_j = scan::skip_space(s, rp_x.at);
    if scan::at(s, rp_j) != 44 { rp_i += 1; continue }
    rp_y = scan::read_number(s, scan::skip_space(s, rp_j + 1));
    if !rp_y.ok { rp_i += 1; continue }
    rp_j = scan::skip_space(s, rp_y.at);
    if scan::at(s, rp_j) != 41 { rp_i += 1; continue }
    rp_j += 1;
    rp_k = scan::skip_space(s, rp_j);
    rp_sm = scan::at(s, rp_k) == 126;
    if rp_sm { rp_k = scan::skip_space(s, rp_k + 1); }
    rp_wv = 0.0;
    rp_has = false;
    if scan::at(s, rp_k) == 64 {
      rp_w = scan::read_number(s, scan::skip_space(s, rp_k + 1));
      if rp_w.ok { rp_wv = rp_w.value; rp_has = true; rp_k = rp_w.at; }
    }
    rp_out.pts += [raster::pt(rp_x.value, rp_y.value)];
    rp_out.smooth += [rp_sm];
    rp_out.widths += [rp_wv];
    rp_out.has_width += [rp_has];
    if rp_has { rp_out.any_width = true; }
    rp_i = rp_k;
  }
  rp_out
}

// ── Geometry ────────────────────────────────────────────────────────

/// The outline of a `Circle`, as `draw.py` builds it: `n` segments round the
/// centre with the FIRST point repeated at the end, so the polygon closes
/// itself and Pillow adds no edge of its own.
///
/// `flat` squashes the vertical radius; the `W/H` factor keeps the circle round
/// on non-square paper, since the coordinates are fractions of each axis.
pub fn circle_pts(cx: float, cy: float, r: float, n: integer, flat: float,
                  pw: integer, ph: integer) -> vector<raster::Pt> {
  cp_ary = r * (((pw as float) / (ph as float)) ?? 1.0) * (1.0 - flat);
  cp_out: vector<raster::Pt> = [];
  for i in 0..=n {
    cp_a = ((2.0 * PI * (i as float)) / (n as float)) ?? 0.0;
    cp_out += [raster::pt(cx + r * cos(cp_a), cy + cp_ary * sin(cp_a))];
  }
  cp_out
}

fn any_smooth(flags: const vector<boolean>) -> boolean {
  for f in flags { if f { return true } }
  false
}

/// The control point `i`, clamped on an open outline and wrapped on a closed one.
fn ctrl(pts: const vector<raster::Pt>, i: integer, closed: boolean) -> raster::Pt {
  cn = len(pts);
  if closed { return pts[i.floor_mod(cn) ?? 0]? }
  ci = i;
  if ci < 0 { ci = 0; }
  if ci > cn - 1 { ci = cn - 1; }
  pts[ci]?
}

/// Half the chord between a point's two neighbours — the Catmull-Rom tangent.
fn half_chord(pts: const vector<raster::Pt>, i: integer, closed: boolean) -> raster::Pt {
  hc_a = ctrl(pts, i - 1, closed);
  hc_b = ctrl(pts, i + 1, closed);
  raster::pt((hc_b.ptx - hc_a.ptx) * 0.5, (hc_b.pty - hc_a.pty) * 0.5)
}

/// Does this outline curve at all?  Fewer than three points, or no `~`, and the
/// control points ARE the polyline — which is what keeps a plain `Poly`
/// unchanged.
pub fn smooth_applies(pts: const vector<raster::Pt>, flags: const vector<boolean>) -> boolean {
  len(pts) >= 3 && any_smooth(flags)
}

/// Expand control points to a dense polyline.
///
/// A point flagged `~` curves (its tangent is half the neighbour chord); a
/// corner uses the segment chord, which makes that segment exactly straight.  So
/// one outline mixes straight runs and curves, and a segment between two corners
/// is a line however many `~` the rest of the outline has.
pub fn smooth_pts(pts: const vector<raster::Pt>, flags: const vector<boolean>,
                  closed: boolean) -> vector<raster::Pt> {
  sp_n = len(pts);
  sp_out: vector<raster::Pt> = [];
  if !smooth_applies(pts, flags) {
    for p in pts { sp_out += [p]; }
    return sp_out
  }
  sp_out += [pts[0]?];
  sp_segs = if closed { sp_n } else { sp_n - 1 };
  for i in 0..sp_segs {
    sp_ia = i % sp_n ?? 0;
    sp_ib = (i + 1) % sp_n ?? 0;
    sp_a = pts[sp_ia]?;
    sp_b = pts[sp_ib]?;
    sp_chx = sp_b.ptx - sp_a.ptx;
    sp_chy = sp_b.pty - sp_a.pty;
    sp_ta = if flags[sp_ia]? { half_chord(pts, sp_ia, closed) } else { raster::pt(sp_chx, sp_chy) };
    sp_tb = if flags[sp_ib]? { half_chord(pts, sp_ib, closed) } else { raster::pt(sp_chx, sp_chy) };
    for k in 1..=SMOOTH_SAMPLES {
      sp_t = ((k as float) / (SMOOTH_SAMPLES as float)) ?? 0.0;
      sp_t2 = sp_t * sp_t;
      sp_t3 = sp_t2 * sp_t;
      sp_h00 = 2.0 * sp_t3 - 3.0 * sp_t2 + 1.0;
      sp_h10 = sp_t3 - 2.0 * sp_t2 + sp_t;
      sp_h01 = -2.0 * sp_t3 + 3.0 * sp_t2;
      sp_h11 = sp_t3 - sp_t2;
      sp_out += [raster::pt(
        sp_h00 * sp_a.ptx + sp_h10 * sp_ta.ptx + sp_h01 * sp_b.ptx + sp_h11 * sp_tb.ptx,
        sp_h00 * sp_a.pty + sp_h10 * sp_ta.pty + sp_h01 * sp_b.pty + sp_h11 * sp_tb.pty)];
    }
  }
  sp_out
}

/// The per-point widths carried across the same expansion, interpolated linearly.
pub fn smooth_vals(vals: const vector<float>, closed: boolean) -> vector<float> {
  sv_n = len(vals);
  sv_out: vector<float> = [];
  sv_out += [vals[0]?];
  sv_segs = if closed { sv_n } else { sv_n - 1 };
  for i in 0..sv_segs {
    sv_a = vals[i % sv_n ?? 0]?;
    sv_b = vals[(i + 1) % sv_n ?? 0]?;
    for k in 1..=SMOOTH_SAMPLES {
      sv_out += [sv_a + (sv_b - sv_a) * (((k as float) / (SMOOTH_SAMPLES as float)) ?? 0.0)];
    }
  }
  sv_out
}

// ── Commands ────────────────────────────────────────────────────────

/// What one command line produced: whether the command matched at all, whether
/// it matched but is not a legal mark, and the points it covers.
///
/// The middle case is the one worth having a flag for.  `Poly (0.1,0.2)
/// rgb=1,2,3` IS a `Poly` — it just cannot be filled with one point — and
/// reporting it as "no command accepted this line" would send the author
/// looking for a typo that is not there.
struct Mark {
  matched: boolean = false,
  bad: boolean = false,
  pts: vector<raster::Pt>,
}

fn no_mark() -> Mark { Mark { matched: false, bad: false, pts: [] } }

/// `Circle\s*\(\s*(n)\s*,\s*(n)\s*\)\s*r=(n)(?:\s+n=(\d+))?(?:\s+flat=(n))?`
fn parse_circle(sc: Sketch, s: text) -> Mark {
  pc_from = 0;
  while true {
    pc_w = scan::find_word_from(s, "circle", pc_from);
    if !pc_w.ok { return no_mark() }
    pc_from = pc_w.at;
    pc_i = scan::skip_space(s, pc_w.at);
    if scan::at(s, pc_i) != 40 { continue }
    pc_x = scan::read_number(s, scan::skip_space(s, pc_i + 1));
    if !pc_x.ok { continue }
    pc_i = scan::skip_space(s, pc_x.at);
    if scan::at(s, pc_i) != 44 { continue }
    pc_y = scan::read_number(s, scan::skip_space(s, pc_i + 1));
    if !pc_y.ok { continue }
    pc_i = scan::skip_space(s, pc_y.at);
    if scan::at(s, pc_i) != 41 { continue }
    pc_i = scan::skip_space(s, pc_i + 1);
    if scan::lower_byte(scan::at(s, pc_i)) != 114 { continue }
    if scan::at(s, pc_i + 1) != 61 { continue }
    pc_r = scan::read_number(s, pc_i + 2);
    if !pc_r.ok { continue }
    pc_i = pc_r.at;
    pc_n = 28;
    pc_j = scan::skip_space(s, pc_i);
    if pc_j > pc_i && scan::lower_byte(scan::at(s, pc_j)) == 110 && scan::at(s, pc_j + 1) == 61 {
      pc_cnt = scan::read_uint(s, pc_j + 2);
      if pc_cnt.ok { pc_n = pc_cnt.value as integer; pc_i = pc_cnt.at; }
    }
    pc_flat = 0.0;
    pc_j = scan::skip_space(s, pc_i);
    if pc_j > pc_i && scan::matches_at(s, pc_j, "flat") && scan::at(s, pc_j + 4) == 61 {
      pc_f = scan::read_number(s, pc_j + 5);
      if pc_f.ok { pc_flat = pc_f.value; }
    }
    pc_pts = circle_pts(pc_x.value, pc_y.value, pc_r.value, pc_n, pc_flat, sc.sw, sc.sh);
    pc_paint = read_paint(s);
    if pc_paint.pk == Stroked {
      sc.ops += [Op { kind: Stroke, pts: pc_pts, paint: pc_paint, widths: [],
                      w: read_width(s, 3), color: read_stroke_colour(s), color2: 0 }];
    } else {
      sc.ops += [Op { kind: Fill, pts: pc_pts, paint: pc_paint, widths: [],
                      w: 3, color: 0, color2: 0 }];
    }
    return Mark { matched: true, bad: false, pts: pc_pts }
  }
  no_mark()
}

/// `Line\s*\(\s*(n)\s*,\s*(n)\s*\)\s*(?:@\s*(n))?\s*-\s*\(…\)\s*(?:@\s*(n))?\s*(?:w\s*=\s*(\d+))?`
fn parse_line_cmd(sc: Sketch, s: text) -> Mark {
  pl_from = 0;
  while true {
    pl_w = scan::find_word_from(s, "line", pl_from);
    if !pl_w.ok { return no_mark() }
    pl_from = pl_w.at;
    pl_i = scan::skip_space(s, pl_w.at);
    if scan::at(s, pl_i) != 40 { continue }
    pl_x0 = scan::read_number(s, scan::skip_space(s, pl_i + 1));
    if !pl_x0.ok { continue }
    pl_i = scan::skip_space(s, pl_x0.at);
    if scan::at(s, pl_i) != 44 { continue }
    pl_y0 = scan::read_number(s, scan::skip_space(s, pl_i + 1));
    if !pl_y0.ok { continue }
    pl_i = scan::skip_space(s, pl_y0.at);
    if scan::at(s, pl_i) != 41 { continue }
    pl_i += 1;
    pl_wa = read_at_width(s, pl_i);
    pl_i = pl_wa.at;
    pl_i = scan::skip_space(s, pl_i);
    // `[-\d.]+` swallows the `-` separator when no space precedes it, and the
    // regex backtracks off it.  Retract the same way rather than refusing.
    if scan::at(s, pl_i) != 45 && pl_wa.ok && scan::at(s, pl_i - 1) == 45 { pl_i -= 1; }
    if scan::at(s, pl_i) != 45 { continue }
    pl_i = scan::skip_space(s, pl_i + 1);
    if scan::at(s, pl_i) != 40 { continue }
    pl_x1 = scan::read_number(s, scan::skip_space(s, pl_i + 1));
    if !pl_x1.ok { continue }
    pl_i = scan::skip_space(s, pl_x1.at);
    if scan::at(s, pl_i) != 44 { continue }
    pl_y1 = scan::read_number(s, scan::skip_space(s, pl_i + 1));
    if !pl_y1.ok { continue }
    pl_i = scan::skip_space(s, pl_y1.at);
    if scan::at(s, pl_i) != 41 { continue }
    pl_wb = read_at_width(s, pl_i + 1);
    pl_base = read_width(s, 3);
    pl_pts: vector<raster::Pt> = [raster::pt(pl_x0.value, pl_y0.value),
                                  raster::pt(pl_x1.value, pl_y1.value)];
    pl_widths: vector<float> = [];
    if pl_wa.ok || pl_wb.ok {
      pl_widths += [if pl_wa.ok { pl_wa.value } else { pl_base as float }];
      pl_widths += [if pl_wb.ok { pl_wb.value } else { pl_base as float }];
    }
    sc.ops += [Op { kind: Stroke, pts: pl_pts, paint: Paint { pk: Stroked, c1: 0, c2: 0, spec: [] },
                    widths: pl_widths, w: pl_base, color: read_stroke_colour(s), color2: 0 }];
    return Mark { matched: true, bad: false, pts: pl_pts }
  }
  no_mark()
}

/// `\s*(?:@\s*([-\d.]+))?` — an endpoint's own pen width.
fn read_at_width(s: text, i: integer) -> Scan {
  ra_j = scan::skip_space(s, i);
  if scan::at(s, ra_j) != 64 { return Scan { ok: false, at: i, value: 0.0 } }
  ra_n = scan::read_number(s, scan::skip_space(s, ra_j + 1));
  if !ra_n.ok { return Scan { ok: false, at: i, value: 0.0 } }
  ra_n
}

/// `Poly (x,y)… [w=N] [<fill>]` — the workhorse: filled when it names a fill,
/// a pen stroke when it does not.
fn parse_poly(sc: Sketch, s: text) -> Mark {
  pp_raw = read_points(s);
  pp_paint = read_paint(s);
  pp_need = if pp_paint.pk == Stroked { 2 } else { 3 };
  if len(pp_raw.pts) < pp_need { return Mark { matched: true, bad: true, pts: [] } }
  if pp_paint.pk != Stroked {
    pp_pts = smooth_pts(pp_raw.pts, pp_raw.smooth, true);
    sc.ops += [Op { kind: Fill, pts: pp_pts, paint: pp_paint, widths: [], w: 3,
                    color: 0, color2: 0 }];
    return Mark { matched: true, bad: false, pts: pp_pts }
  }
  pp_base = read_width(s, 3);
  pp_ink = read_stroke_colour(s);
  if pp_raw.any_width {
    pp_vals: vector<float> = [];
    for i in 0..len(pp_raw.pts) {
      pp_vals += [if pp_raw.has_width[i]? { pp_raw.widths[i]? } else { pp_base as float }];
    }
    pp_open = smooth_pts(pp_raw.pts, pp_raw.smooth, false);
    pp_ow = if smooth_applies(pp_raw.pts, pp_raw.smooth) { smooth_vals(pp_vals, false) }
            else { pp_vals };
    sc.ops += [Op { kind: Stroke, pts: pp_open, paint: pp_paint, widths: pp_ow,
                    w: pp_base, color: pp_ink, color2: 0 }];
    return Mark { matched: true, bad: false, pts: pp_open }
  }
  pp_line = smooth_pts(pp_raw.pts, pp_raw.smooth, false);
  sc.ops += [Op { kind: Stroke, pts: pp_line, paint: pp_paint, widths: [],
                  w: pp_base, color: pp_ink, color2: 0 }];
  Mark { matched: true, bad: false, pts: pp_line }
}

/// Where `name`'s box lives, creating it on first mention.
fn elem_index(sc: Sketch, nm: text) -> integer {
  for i in 0..len(sc.elems) {
    if sc.elems[i]?.ename == nm { return i }
  }
  sc.elems += [Elem { ename: nm, seen: false, bx0: 0.0, by0: 0.0, bx1: 0.0, by1: 0.0 }];
  len(sc.elems) - 1
}

/// Grow the current element's box to hold these points.
fn acc_pts(sc: Sketch, idx: integer, pts: const vector<raster::Pt>) {
  if idx < 0 || len(pts) == 0 { return }
  ap_e = sc.elems[idx]?;
  ap_seen = ap_e.seen;
  ap_x0 = ap_e.bx0;
  ap_y0 = ap_e.by0;
  ap_x1 = ap_e.bx1;
  ap_y1 = ap_e.by1;
  for p in pts {
    if !ap_seen {
      ap_x0 = p.ptx; ap_y0 = p.pty; ap_x1 = p.ptx; ap_y1 = p.pty;
      ap_seen = true;
    } else {
      if p.ptx < ap_x0 { ap_x0 = p.ptx; }
      if p.pty < ap_y0 { ap_y0 = p.pty; }
      if p.ptx > ap_x1 { ap_x1 = p.ptx; }
      if p.pty > ap_y1 { ap_y1 = p.pty; }
    }
  }
  sc.elems[idx] = Elem { ename: ap_e.ename, seen: ap_seen,
                         bx0: ap_x0, by0: ap_y0, bx1: ap_x1, by1: ap_y1 };
}

/// `Background …` — transparent, a colour ramp, or a grey ramp.
fn parse_background(sc: Sketch, s: text, low: text) -> boolean {
  if low.contains("transparent") || low.contains("none") {
    sc.transparent = true;
    return true
  }
  pb_tc = scan::find_option(s, "topc");
  pb_bc = scan::find_option(s, "botc");
  if pb_tc.ok && pb_bc.ok {
    pb_t = read_rgb_at(s, pb_tc.at);
    pb_b = read_rgb_at(s, pb_bc.at);
    if pb_t.ok && pb_b.ok {
      sc.ops += [Op { kind: Sky, pts: [], paint: Paint { pk: Stroked, c1: 0, c2: 0, spec: [] },
                      widths: [], w: 3, color: pb_t.color, color2: pb_b.color }];
      return true
    }
  }
  pb_gt = scan::find_option(s, "top");
  if !pb_gt.ok { return false }
  pb_gb = scan::find_option(s, "bottom");
  if !pb_gb.ok { pb_gb = scan::find_option(s, "bot"); }
  if !pb_gb.ok { return false }
  pb_tv = scan::read_unsigned(s, pb_gt.at);
  pb_bv = scan::read_unsigned(s, pb_gb.at);
  if !pb_tv.ok || !pb_bv.ok { return false }
  sc.ops += [Op { kind: Sky, pts: [], paint: Paint { pk: Stroked, c1: 0, c2: 0, spec: [] },
                  widths: [], w: 3, color: grey(pb_tv.value), color2: grey(pb_bv.value) }];
  true
}

/// `size WxH` — a FULL-line match, so nothing else can be on it.
fn parse_size(sc: Sketch, s: text) -> boolean {
  if !scan::matches_at(s, 0, "size") { return false }
  pz_i = scan::skip_space(s, 4);
  if pz_i == 4 { return false }
  pz_w = scan::read_uint(s, pz_i);
  if !pz_w.ok { return false }
  pz_i = scan::skip_space(s, pz_w.at);
  if scan::lower_byte(scan::at(s, pz_i)) != 120 { return false }
  pz_h = scan::read_uint(s, scan::skip_space(s, pz_i + 1));
  if !pz_h.ok || pz_h.at != size(s) { return false }
  sc.sw = pz_w.value as integer;
  sc.sh = pz_h.value as integer;
  true
}

/// `landmark <name> = <value>` — anchored at the start of the line.
fn parse_landmark(sc: Sketch, s: text) -> boolean {
  pm_i = scan::skip_space(s, 8);
  if pm_i == 8 { return false }
  pm_j = pm_i;
  while scan::is_word_byte(scan::at(s, pm_j)) { pm_j += 1; }
  if pm_j == pm_i { return false }
  pm_k = scan::skip_space(s, pm_j);
  if scan::at(s, pm_k) != 61 { return false }
  pm_v = scan::read_number(s, scan::skip_space(s, pm_k + 1));
  if !pm_v.ok { return false }
  sc.landmarks += [Landmark { lname: s[pm_i..pm_j], lvalue: pm_v.value }];
  true
}

/// Parse a whole `.draw` source.
///
/// The command order below is `draw.py`'s, and it is not alphabetical or
/// tidy — `Circle` is tried before `Poly`, `Line` last of all.  Keep it: a
/// grammar this loose is defined by the order its alternatives are tried in,
/// and re-sorting it silently re-reads lines that sit in two alternatives.
pub fn parse_scene(src: text) -> Sketch {
  sc = Sketch { sw: 800, sh: 800, transparent: false, ops: [], elems: [], landmarks: [],
               checks: [], unparsed: [], deferred: [] };
  ps_cur = -1;
  ps_no = 0;
  for raw in src.split('\n') {
    ps_no += 1;
    ps_s = raw.trim();
    if ps_s == "" || ps_s.starts_with("#") { continue }
    ps_low = ps_s.to_lowercase();
    if ps_low.starts_with("name ") {
      ps_cur = elem_index(sc, ps_s[5..size(ps_s)].trim());
      continue
    }
    if ps_low.starts_with("landmark") {
      if !parse_landmark(sc, ps_s) { sc.unparsed += ["line {ps_no}: {ps_s}"]; }
      continue
    }
    if ps_low.starts_with("check") {
      sc.checks += [ps_s[5..size(ps_s)].trim()];
      continue
    }
    if ps_low.starts_with("background") {
      if !parse_background(sc, ps_s, ps_low) { sc.unparsed += ["line {ps_no}: {ps_s}"]; }
      continue
    }
    if parse_size(sc, ps_s) { continue }
    ps_m = parse_circle(sc, ps_s);
    if ps_m.matched { acc_pts(sc, ps_cur, ps_m.pts); continue }
    // `Petals` and `Fronds` are recognised here so they cannot fall through to
    // `Line` and be read as something else — and listed, so their absence from
    // the picture is a fact the caller has rather than one they discover.
    if ps_low.starts_with("petals") { sc.deferred += ["line {ps_no}: Petals"]; continue }
    if ps_low.starts_with("fronds") { sc.deferred += ["line {ps_no}: Fronds"]; continue }
    if ps_low.starts_with("poly") {
      ps_p = parse_poly(sc, ps_s);
      if ps_p.bad { sc.unparsed += ["line {ps_no}: {ps_s}"]; }
      else { acc_pts(sc, ps_cur, ps_p.pts); }
      continue
    }
    ps_l = parse_line_cmd(sc, ps_s);
    if ps_l.matched { acc_pts(sc, ps_cur, ps_l.pts); continue }
    sc.unparsed += ["line {ps_no}: {ps_s}"];
  }
  sc
}

/// Paper fractions to pixels on the supersampled canvas.
fn to_pixels(pts: const vector<raster::Pt>, bw: integer, bh: integer) -> vector<raster::Pt> {
  tp_out: vector<raster::Pt> = [];
  for p in pts { tp_out += [raster::pt(p.ptx * (bw as float), p.pty * (bh as float))]; }
  tp_out
}

/// The vertical background ramp, one Bresenham row at a time — which is what
/// `draw.py` does, and the row-by-row truncation of the interpolated colour is
/// visible in the result.
fn draw_sky(cv: graphics::Canvas, top: integer, bot: integer, bw: integer, bh: integer) {
  dk_den = if bh - 1 < 1 { 1 } else { bh - 1 };
  dk_tr = graphics::color_r(top);
  dk_tg = graphics::color_g(top);
  dk_tb = graphics::color_b(top);
  dk_dr = graphics::color_r(bot) - dk_tr;
  dk_dg = graphics::color_g(bot) - dk_tg;
  dk_db = graphics::color_b(bot) - dk_tb;
  for y in 0..bh {
    dk_t = ((y as float) / (dk_den as float)) ?? 0.0;
    dk_c = graphics::rgba(((dk_tr as float) + (dk_dr as float) * dk_t) as integer,
                          ((dk_tg as float) + (dk_dg as float) * dk_t) as integer,
                          ((dk_tb as float) + (dk_db as float) * dk_t) as integer, 255);
    dk_row: vector<raster::Pt> = [raster::pt(0.0, y as float), raster::pt(bw as float, y as float)];
    raster::thin_line(cv, dk_row, dk_c);
  }
}

fn draw_fill(cv: graphics::Canvas, op: const Op, bw: integer, bh: integer) {
  if op.paint.pk == Stroked { return }
  df_px = to_pixels(op.pts, bw, bh);
  if op.paint.pk == Solid {
    raster::fill_poly(cv, df_px, op.paint.c1);
    return
  }
  fill_gradient(cv, op, df_px, bw, bh);
}

/// The side of the gradient grid `draw.py` computes on.  Not a quality knob: the
/// picture is this grid ENLARGED by a bicubic filter, so changing it changes
/// every gradient in the corpus.
const GRADIENT_GRID = 100;

/// Pillow's `or`-idiom guard: a zero denominator becomes this rather than
/// dividing by zero.  It is a real number in the arithmetic, not a sentinel — a
/// degenerate axis therefore saturates `t` instead of blanking the shape.
const GRADIENT_EPS = 0.000000001;

/// `draw.py`'s `_make_gradient`: the ramp computed small, then ENLARGED.
///
/// ⚠ The enlargement is the contract, not an optimisation.  `Image.resize(size)`
/// with no `resample=` is BICUBIC, whose negative lobes overshoot — a 100x100
/// ramp blown up to a shape's bounding box carries ringing that computing the
/// ramp directly at full size would not have. Reproduce the two steps or the
/// pixels differ.
fn make_gradient(pa: const Paint, fx0: float, fy0: float, fx1: float, fy1: float,
                 gw: integer, gh: integer) -> graphics::Canvas {
  // Defaults, when no `dir=` / `at=` was given: a linear ramp runs top to bottom
  // over the bounding box, a radial one starts at its centre.
  mg_ax = (fx0 + fx1) / 2.0;
  mg_ay = fy0;
  mg_bx = (fx0 + fx1) / 2.0;
  mg_by = fy1;
  mg_cx = (fx0 + fx1) / 2.0;
  mg_cy = (fy0 + fy1) / 2.0;
  mg_r = (if fx1 - fx0 > fy1 - fy0 { fx1 - fx0 } else { fy1 - fy0 }) / 2.0;
  if pa.pk == Linear && len(pa.spec) == 4 {
    mg_ax = pa.spec[0]?; mg_ay = pa.spec[1]?; mg_bx = pa.spec[2]?; mg_by = pa.spec[3]?;
  }
  if pa.pk == Radial && len(pa.spec) == 3 {
    mg_cx = pa.spec[0]?; mg_cy = pa.spec[1]?; mg_r = pa.spec[2]?;
  }
  mg_dx = mg_bx - mg_ax;
  mg_dy = mg_by - mg_ay;
  mg_den = mg_dx * mg_dx + mg_dy * mg_dy;
  if mg_den == 0.0 { mg_den = GRADIENT_EPS; }
  if mg_r == 0.0 { mg_r = GRADIENT_EPS; }
  mg_r1 = graphics::color_r(pa.c1); mg_g1 = graphics::color_g(pa.c1); mg_b1 = graphics::color_b(pa.c1);
  mg_r2 = graphics::color_r(pa.c2); mg_g2 = graphics::color_g(pa.c2); mg_b2 = graphics::color_b(pa.c2);
  mg_small = graphics::canvas(GRADIENT_GRID, GRADIENT_GRID, 0);
  mg_g = GRADIENT_GRID as float;
  for j in 0..GRADIENT_GRID {
    mg_fy = fy0 + (fy1 - fy0) * (((j as float) + 0.5) / mg_g ?? 0.0);
    for i in 0..GRADIENT_GRID {
      mg_fx = fx0 + (fx1 - fx0) * (((i as float) + 0.5) / mg_g ?? 0.0);
      mg_t = if pa.pk == Linear {
        (((mg_fx - mg_ax) * mg_dx + (mg_fy - mg_ay) * mg_dy) / mg_den ?? 0.0)
      } else {
        (sqrt((mg_fx - mg_cx) * (mg_fx - mg_cx) + (mg_fy - mg_cy) * (mg_fy - mg_cy)) / mg_r ?? 0.0)
      };
      if mg_t < 0.0 { mg_t = 0.0; }
      if mg_t > 1.0 { mg_t = 1.0; }
      mg_small.set_pixel(i, j, graphics::rgba(
        ((mg_r1 as float) + ((mg_r2 - mg_r1) as float) * mg_t) as integer,
        ((mg_g1 as float) + ((mg_g2 - mg_g1) as float) * mg_t) as integer,
        ((mg_b1 as float) + ((mg_b2 - mg_b1) as float) * mg_t) as integer, 255));
    }
  }
  mg_small.resize_bicubic(if gw < 1 { 1 } else { gw }, if gh < 1 { 1 } else { gh })
}

/// Fill a shape with a gradient: build the ramp over the shape's bounding box,
/// then stamp it through the shape as a mask.
///
/// ⚠ The mask is a POLYGON FILL on a zero-filled buffer, so it only ever holds
/// "set" or "unset" — measured, `ImageDraw.polygon` on an `L` image produces
/// exactly {0, 255} and nothing between. That is what makes the paste a hard
/// SELECT rather than an alpha blend, and why the gradient's own edge is not
/// anti-aliased here: the 3x supersample is what smooths it, one step later.
fn fill_gradient(cv: graphics::Canvas, op: const Op, pxs: const vector<raster::Pt>,
                 bw: integer, bh: integer) {
  fg_n = len(op.pts);
  if fg_n == 0 { return }
  fg_x0 = op.pts[0]?.ptx; fg_y0 = op.pts[0]?.pty;
  fg_x1 = fg_x0; fg_y1 = fg_y0;
  for p in op.pts {
    if p.ptx < fg_x0 { fg_x0 = p.ptx; }
    if p.pty < fg_y0 { fg_y0 = p.pty; }
    if p.ptx > fg_x1 { fg_x1 = p.ptx; }
    if p.pty > fg_y1 { fg_y1 = p.pty; }
  }
  fg_px0 = (fg_x0 * (bw as float)) as integer;
  fg_py0 = (fg_y0 * (bh as float)) as integer;
  fg_w = ((fg_x1 * (bw as float)) as integer) + 1 - fg_px0;
  fg_h = ((fg_y1 * (bh as float)) as integer) + 1 - fg_py0;
  if fg_w < 1 { fg_w = 1; }
  if fg_h < 1 { fg_h = 1; }
  fg_grad = make_gradient(op.paint, fg_x0, fg_y0, fg_x1, fg_y1, fg_w, fg_h);
  fg_mask = graphics::canvas(fg_w, fg_h, 0);
  fg_local: vector<raster::Pt> = [];
  for p in pxs { fg_local += [raster::pt(p.ptx - (fg_px0 as float), p.pty - (fg_py0 as float))]; }
  raster::fill_poly(fg_mask, fg_local, graphics::rgba(255, 255, 255, 255));
  for y in 0..fg_h {
    for x in 0..fg_w {
      if graphics::color_a(fg_mask.get_pixel(x, y)) != 0 {
        cv.set_pixel(fg_px0 + x, fg_py0 + y, fg_grad.get_pixel(x, y));
      }
    }
  }
}

fn draw_stroke(cv: graphics::Canvas, op: const Op, bw: integer, bh: integer) {
  ds_px = to_pixels(op.pts, bw, bh);
  if len(ds_px) < 2 { return }
  if len(op.widths) > 0 { ribbon(cv, ds_px, op.widths, op.color); return }
  ds_w = op.w * SUPERSAMPLE;
  if ds_w < 1 { ds_w = 1; }
  for i in 0..(len(ds_px) - 1) {
    ds_a = ds_px[i]?;
    ds_b = ds_px[i + 1]?;
    if ds_w <= 1 {
      ds_seg: vector<raster::Pt> = [ds_a, ds_b];
      raster::thin_line(cv, ds_seg, op.color);
    } else {
      raster::wide_line(cv, ds_a.ptx, ds_a.pty, ds_b.ptx, ds_b.pty, op.color, ds_w);
    }
  }
}

/// A stroke whose pen width varies along it, drawn as a filled band.
///
/// Offsetting each point along its own normal by half its own width and filling
/// between the two sides is what lets a hair strand be thick at the root and
/// thin at the tip — a pen cannot do that, and a polygon can.
fn ribbon(cv: graphics::Canvas, pxs: const vector<raster::Pt>, widths: const vector<float>,
          color: integer) {
  rb_n = len(pxs);
  if rb_n < 2 { return }
  rb_left: vector<raster::Pt> = [];
  rb_right: vector<raster::Pt> = [];
  for i in 0..rb_n {
    rb_hw = widths[i]? * (SUPERSAMPLE as float) / 2.0;
    if rb_hw < 0.5 { rb_hw = 0.5; }
    rb_p = pxs[i]?;
    rb_tx = 0.0;
    rb_ty = 0.0;
    if i == 0 {
      rb_q = pxs[1]?;
      rb_tx = rb_q.ptx - rb_p.ptx;
      rb_ty = rb_q.pty - rb_p.pty;
    } else if i == rb_n - 1 {
      rb_q = pxs[i - 1]?;
      rb_tx = rb_p.ptx - rb_q.ptx;
      rb_ty = rb_p.pty - rb_q.pty;
    } else {
      rb_a = pxs[i + 1]?;
      rb_b = pxs[i - 1]?;
      rb_tx = rb_a.ptx - rb_b.ptx;
      rb_ty = rb_a.pty - rb_b.pty;
    }
    rb_l = sqrt(rb_tx * rb_tx + rb_ty * rb_ty);
    if rb_l == 0.0 { rb_l = 1.0; }
    rb_nx = (-rb_ty / rb_l) ?? 0.0;
    rb_ny = (rb_tx / rb_l) ?? 0.0;
    rb_left += [raster::pt(rb_p.ptx + rb_nx * rb_hw, rb_p.pty + rb_ny * rb_hw)];
    rb_right += [raster::pt(rb_p.ptx - rb_nx * rb_hw, rb_p.pty - rb_ny * rb_hw)];
  }
  rb_poly: vector<raster::Pt> = [];
  for p in rb_left { rb_poly += [p]; }
  for i in 0..rb_n { rb_poly += [rb_right[rb_n - 1 - i]?]; }
  raster::fill_poly(cv, rb_poly, color);
}

/// Draw a parsed scene: at `SUPERSAMPLE`x, then resampled down with Lanczos.
///
/// A transparent background renders on transparent pixels so the resample
/// anti-aliases edges against ALPHA rather than against white — which is what
/// makes the result a sprite that can be composited over anything, with no
/// colour-key hack.
pub fn render(sc: const Sketch) -> graphics::Canvas {
  rn_bw = sc.sw * SUPERSAMPLE;
  rn_bh = sc.sh * SUPERSAMPLE;
  rn_cv = graphics::canvas(rn_bw, rn_bh,
                           if sc.transparent { 0 } else { graphics::rgba(255, 255, 255, 255) });
  for op in sc.ops {
    match op.kind {
      Sky => draw_sky(rn_cv, op.color, op.color2, rn_bw, rn_bh),
      Fill => draw_fill(rn_cv, op, rn_bw, rn_bh),
      Stroke => draw_stroke(rn_cv, op, rn_bw, rn_bh),
    }
  }
  rn_cv.resize_lanczos(sc.sw, sc.sh)
}

/// Render a `.draw` file to a PNG, answering the parsed scene so the caller can
/// report what it could not accept.
pub fn render_file(src: text, out: text) -> Sketch {
  rf_sc = parse_scene(file(src).content()?);
  rf_cv = render(rf_sc);
  _ = rf_cv.save_png(out);
  rf_sc
}

// Copyright (c) 2026 Jurjen Stellingwerff
// SPDX-License-Identifier: LGPL-3.0-or-later
//
// raster — Pillow's polygon and thick-line rasteriser, reproduced exactly.
//
// ── ⚠ WHY THIS IS NOT `graphics::fill_polygon` ──────────────────────
//
// `graphics` already fills polygons, and this deliberately does not call it.
// The two answer different questions:
//
//   * `graphics::fill_polygon` agrees with `graphics::fill_triangle`, pixel for
//     pixel, on every triangle — a gated, published contract.  It interpolates
//     each crossing in INTEGER arithmetic and fills an inclusive span between a
//     pair of them.
//   * this fills the pixels whose CENTRES fall inside the outline, in C `float`,
//     which is what `ImageDraw.polygon` does.
//
// They are different rules and they disagree constantly: on 400 random polygons
// the two agree on **4**, and on `graphics`' own reference triangle they differ
// by 35 pixels.  A `.draw` scene is specified by what `draw.py` renders, and
// `draw.py`'s filler *is* Pillow — so a renderer that wants to be diffed against
// the corpus has to sign Pillow's rule, not a better one.
//
// ── ⚠ THE `single` IS LOAD-BEARING ──────────────────────────────────
//
// `Edge.dx` and the crossing array are C `float` (32-bit) in `Draw.c`, and the
// two exact-equality tests below — "is this crossing an integer?" and "do these
// two edges cross at the same x?" — read that rounding.  Widen them to `float`
// and 12 of 500 random polygons rasterise differently.  Every value that is a C
// `float` there is a `single` here, and the arithmetic is per-operation binary32
// on both backends.
//
// Ported from Pillow 10.2.0, `src/libImaging/Draw.c` + `src/_imaging.c`.
use graphics;

/// A point in canvas pixels, before Pillow truncates it to a whole pixel.
///
/// The truncation is `(int)` on a C double — toward ZERO, so `-0.4` lands on 0
/// and not on -1.  Keep coordinates fractional right up to the rasteriser and
/// the port and the oracle round the same way.
pub struct Pt {
  ptx: float,
  pty: float,
}

/// Build a point without spelling the field names.
pub fn pt(px: float, py: float) -> Pt {
  Pt { ptx: px, pty: py }
}

/// One outline edge, field for field as `Edge` in `Draw.c`.
struct Edge {
  exmin: integer,
  exmax: integer,
  eymin: integer,
  eymax: integer,
  edx: single,
  ex0: integer,
  ey0: integer,
}

/// `ROUND_UP` from `Draw.c` — half AWAY from zero.
///
/// Not `floor(f + 0.5)`: the sign lives outside, so the negative side mirrors
/// the positive one exactly (`ROUND_UP(-0.5)` is -1, not 0).
pub fn round_up(ru_v: float) -> integer {
  if ru_v >= 0.0 { floor(ru_v + 0.5) as integer }
  else { -(floor(abs(ru_v) + 0.5) as integer) }
}

/// `ROUND_DOWN` from `Draw.c` — `ceil(f - 0.5)`, mirrored about zero.
///
/// The mirroring is the whole subtlety: `ROUND_DOWN(-0.45)` is **0**, because
/// the `ceil` runs on `|f| - 0.5` and the sign is applied afterwards.  Take the
/// obvious `ceil(-0.95)` instead and a span that should end on column 0 ends on
/// column -1, which clips away and draws nothing.
pub fn round_down(rd_v: float) -> integer {
  if rd_v >= 0.0 { ceil(rd_v - 0.5) as integer }
  else { -(ceil(abs(rd_v) - 0.5) as integer) }
}

/// C `roundf` on a 32-bit float: half away from zero, answer stays binary32.
fn roundf(rf_v: single) -> single {
  rf_d = rf_v as float;
  if rf_d >= 0.0 { floor(rf_d + 0.5) as single } else { ceil(rf_d - 0.5) as single }
}

/// `add_edge` from `Draw.c`.
fn make_edge(mx0: integer, my0: integer, mx1: integer, my1: integer) -> Edge {
  Edge {
    exmin: if mx0 <= mx1 { mx0 } else { mx1 },
    exmax: if mx0 <= mx1 { mx1 } else { mx0 },
    eymin: if my0 <= my1 { my0 } else { my1 },
    eymax: if my0 <= my1 { my1 } else { my0 },
    edx: if my0 == my1 { 0.0f }
         else { (((mx1 - mx0) as single) / ((my1 - my0) as single) ?? 0.0f) },
    ex0: mx0,
    ey0: my0,
  }
}

/// Where an edge meets scanline `y`, in the `float` `Draw.c` carries it in.
fn edge_x(exe: const Edge, exy: integer) -> single {
  ((exy - exe.ey0) as single) * exe.edx + (exe.ex0 as single)
}

/// `hline32` from `Draw.c`: an INCLUSIVE span `[x0, x1]`, clipped to the canvas.
///
/// It STORES the ink rather than compositing it — `ImageDraw` on an RGBA image
/// replaces the pixel, alpha included, which is why a half-transparent mark does
/// not show what was under it.
fn pil_hline(cv: graphics::Canvas, hx0: integer, hy: integer, hx1: integer, ink: integer) {
  if hy < 0 || hy >= cv.height { return }
  hl_lo = hx0;
  hl_hi = hx1;
  if hl_lo < 0 { hl_lo = 0; } else if hl_lo >= cv.width { return }
  if hl_hi < 0 { return } else if hl_hi >= cv.width { hl_hi = cv.width - 1; }
  if hl_lo > hl_hi { return }
  hl_d = &cv.data;
  hl_base = hy * cv.width;
  for hl_x in hl_lo..=hl_hi {
    hl_d[hl_base + hl_x] = ink;
  }
}

/// `polygon_generic` from `Draw.c`, in the non-blending 32-bit form
/// `ImageDraw.Draw(rgba_image)` selects.
///
/// Three things here look like mistakes and are the contract:
///
///  * a HORIZONTAL edge is drawn immediately, as a span from its own `xmin` to
///    its own `xmax`, and takes no part in the scanline crossings;
///  * an edge ending on this row contributes its crossing TWICE (unless the row
///    is the polygon's last), which is how a vertex the outline passes through
///    keeps the parity right;
///  * the "connect discontiguous corners" repair writes back to `xx[k]` where
///    `k` is an EDGE index, not a crossing index — so it can move a crossing
///    belonging to a different edge.  Reproduced verbatim, because the goldens
///    were rendered through it.
///
/// ⚠ It scores 124 for control-flow complexity and that is deliberate: this is a
/// line-for-line port of one C function, and the thing a reader needs to be able to
/// do is put the two side by side.  Splitting it would make the correspondence
/// unverifiable and would buy nothing — there is no second caller and no variation
/// to factor out.
fn polygon_generic(cv: graphics::Canvas, edges: const vector<Edge>, ink: integer) {
  if len(edges) <= 0 { return }
  pg_table: vector<Edge> = [];
  pg_ymin = cv.height - 1;
  pg_ymax = 0;
  for e in edges {
    if pg_ymin > e.eymin { pg_ymin = e.eymin; }
    if pg_ymax < e.eymax { pg_ymax = e.eymax; }
    if e.eymin == e.eymax {
      pil_hline(cv, e.exmin, e.eymin, e.exmax, ink);
      continue
    }
    pg_table += [e];
  }
  if pg_ymin < 0 { pg_ymin = 0; }
  if pg_ymax > cv.height { pg_ymax = cv.height; }
  pg_ec = len(pg_table);
  if pg_ec == 0 { return }
  // One crossing buffer for the whole polygon, exactly as `Draw.c` allocates it:
  // the corner repair reaches back into it by edge index, so it must persist
  // across the row and must be at least two slots per edge.
  pg_xx: vector<single> = [];
  for _ in 0..(pg_ec * 2) { pg_xx += [0.0f]; }
  pg_y = pg_ymin;
  while pg_y <= pg_ymax {
    pg_j = 0;
    for i in 0..pg_ec {
      pg_cur = pg_table[i]?;
      if pg_y >= pg_cur.eymin && pg_y <= pg_cur.eymax {
        pg_xx[pg_j] = edge_x(pg_cur, pg_y);
        pg_j += 1;
        if pg_y == pg_cur.eymax && pg_y < pg_ymax {
          pg_xx[pg_j] = pg_xx[pg_j - 1]?;
          pg_j += 1;
        } else if pg_cur.edx != 0.0f && roundf(pg_xx[pg_j - 1]?) == pg_xx[pg_j - 1]? {
          for k in 0..i {
            pg_oth = pg_table[k]?;
            if (pg_cur.edx > 0.0f && pg_oth.edx <= 0.0f)
               || (pg_cur.edx < 0.0f && pg_oth.edx >= 0.0f) { continue }
            if ((pg_y == pg_cur.eymin && pg_y == pg_oth.eymin)
                || (pg_y == pg_cur.eymax && pg_y == pg_oth.eymax))
               && pg_xx[pg_j - 1]? == edge_x(pg_oth, pg_y) {
              // The row below (or above, on the last row) says which side the
              // two edges leave the corner on.
              pg_off = if pg_y == pg_ymax { -1 } else { 1 };
              pg_ac = edge_x(pg_cur, pg_y + pg_off);
              pg_ao = edge_x(pg_oth, pg_y + pg_off);
              if pg_y == pg_cur.eymax {
                if pg_cur.edx > 0.0f {
                  pg_xx[k] = (if pg_ac > pg_ao { pg_ac } else { pg_ao }) + 1.0f;
                } else {
                  pg_xx[k] = (if pg_ac < pg_ao { pg_ac } else { pg_ao }) - 1.0f;
                }
              } else {
                if pg_cur.edx > 0.0f {
                  pg_xx[k] = if pg_ac < pg_ao { pg_ac } else { pg_ao };
                } else {
                  pg_xx[k] = (if pg_ac > pg_ao { pg_ac } else { pg_ao }) + 1.0f;
                }
              }
              break
            }
          }
        }
      }
    }
    // Insertion sort over `xx[0 .. j)` — a row holds a handful of crossings, and
    // equal ones are interchangeable, so this and `Draw.c`'s `qsort` agree.
    pg_s = 1;
    while pg_s < pg_j {
      pg_v = pg_xx[pg_s]?;
      pg_t = pg_s;
      while pg_t > 0 && pg_xx[pg_t - 1]? > pg_v {
        pg_xx[pg_t] = pg_xx[pg_t - 1]?;
        pg_t -= 1;
      }
      pg_xx[pg_t] = pg_v;
      pg_s += 1;
    }
    pg_p = 1;
    while pg_p < pg_j {
      pil_hline(cv, round_up(pg_xx[pg_p - 1]? as float), pg_y,
                round_down(pg_xx[pg_p]? as float), ink);
      pg_p += 2;
    }
    pg_y += 1;
  }
}

/// Fill a polygon the way `ImageDraw.polygon(pts, fill=…)` does.
///
/// The outline closes itself only when the last point is not already the first —
/// a circle handed in with its start repeated gets no extra edge, which is how
/// `draw.py` passes one.
pub fn fill_poly(cv: graphics::Canvas, pts: const vector<Pt>, ink: integer) {
  fp_count = len(pts);
  if fp_count <= 0 { return }
  fp_xs: vector<integer> = [];
  fp_ys: vector<integer> = [];
  for p in pts {
    fp_xs += [p.ptx as integer];
    fp_ys += [p.pty as integer];
  }
  fp_edges: vector<Edge> = [];
  fp_i = 0;
  while fp_i < fp_count - 1 {
    fp_x0 = fp_xs[fp_i]?;
    fp_y0 = fp_ys[fp_i]?;
    fp_x1 = fp_xs[fp_i + 1]?;
    fp_y1 = fp_ys[fp_i + 1]?;
    fp_merged = false;
    // Two horizontal steps running the same way are ONE span in Pillow, so the
    // second widens the first edge instead of adding its own.
    if fp_y0 == fp_y1 && fp_i != 0 && fp_y0 == fp_ys[fp_i - 1]? && len(fp_edges) > 0 {
      fp_prev = fp_xs[fp_i - 1]?;
      fp_last = len(fp_edges) - 1;
      fp_e = fp_edges[fp_last]?;
      if fp_x1 > fp_x0 && fp_x0 > fp_prev {
        fp_edges[fp_last] = Edge { exmin: fp_e.exmin, exmax: fp_x1, eymin: fp_e.eymin,
                                   eymax: fp_e.eymax, edx: fp_e.edx, ex0: fp_e.ex0,
                                   ey0: fp_e.ey0 };
        fp_merged = true;
      } else if fp_x1 < fp_x0 && fp_x0 < fp_prev {
        fp_edges[fp_last] = Edge { exmin: fp_x1, exmax: fp_e.exmax, eymin: fp_e.eymin,
                                   eymax: fp_e.eymax, edx: fp_e.edx, ex0: fp_e.ex0,
                                   ey0: fp_e.ey0 };
        fp_merged = true;
      }
    }
    if !fp_merged {
      fp_edges += [make_edge(fp_x0, fp_y0, fp_x1, fp_y1)];
    }
    fp_i += 1;
  }
  fp_end = fp_count - 1;
  if fp_xs[fp_end]? != fp_xs[0]? || fp_ys[fp_end]? != fp_ys[0]? {
    fp_edges += [make_edge(fp_xs[fp_end]?, fp_ys[fp_end]?, fp_xs[0]?, fp_ys[0]?)];
  }
  polygon_generic(cv, fp_edges, ink);
}

/// One thick segment, the way `ImageDraw.line(…, width=w)` draws it for `w > 1`.
///
/// Pillow does not stroke — it builds a four-cornered quad around the segment
/// and fills that, with the two half-widths rounded APART (`ROUND_UP` one side,
/// `ROUND_DOWN` the other) so an even width still lands on whole pixels.  A
/// polyline is therefore a series of independent quads with no joins, and the
/// overlap at a bend is what rounds the corner.
pub fn wide_line(cv: graphics::Canvas, wx0: float, wy0: float, wx1: float, wy1: float,
                 ink: integer, wwidth: integer) {
  wl_x0 = wx0 as integer;
  wl_y0 = wy0 as integer;
  wl_x1 = wx1 as integer;
  wl_y1 = wy1 as integer;
  wl_dx = wl_x1 - wl_x0;
  wl_dy = wl_y1 - wl_y0;
  if wl_dx == 0 && wl_dy == 0 {
    cv.set_pixel(wl_x0, wl_y0, ink);
    return
  }
  wl_big = sqrt(((wl_dx * wl_dx + wl_dy * wl_dy) as float));
  wl_small = (((wwidth - 1) as float) / 2.0 ?? 0.0);
  wl_rmax = ((round_up(wl_small) as float) / wl_big ?? 0.0);
  wl_rmin = ((round_down(wl_small) as float) / wl_big ?? 0.0);
  wl_dxmin = round_down(wl_rmin * (wl_dy as float));
  wl_dxmax = round_down(wl_rmax * (wl_dy as float));
  wl_dymin = round_down(wl_rmin * (wl_dx as float));
  wl_dymax = round_down(wl_rmax * (wl_dx as float));
  wl_vx: vector<integer> = [wl_x0 - wl_dxmin, wl_x1 - wl_dxmin,
                            wl_x1 + wl_dxmax, wl_x0 + wl_dxmax];
  wl_vy: vector<integer> = [wl_y0 + wl_dymax, wl_y1 + wl_dymax,
                            wl_y1 - wl_dymin, wl_y0 - wl_dymin];
  wl_edges: vector<Edge> = [];
  for k in 0..4 {
    wl_edges += [make_edge(wl_vx[k]?, wl_vy[k]?, wl_vx[(k + 1) % 4]?, wl_vy[(k + 1) % 4]?)];
  }
  polygon_generic(cv, wl_edges, ink);
}

/// `line32` from `Draw.c` — Bresenham, and it stops one pixel SHORT of the end.
///
/// `draw_lines` puts that missing pixel back after the last segment, which is
/// why this is a polyline call rather than a segment one: split it into
/// segments and each one grows a pixel Pillow never drew.
pub fn thin_line(cv: graphics::Canvas, pts: const vector<Pt>, ink: integer) {
  tl_n = len(pts);
  if tl_n < 2 { return }
  tl_lx = 0;
  tl_ly = 0;
  for i in 0..(tl_n - 1) {
    tl_a = pts[i]?;
    tl_b = pts[i + 1]?;
    tl_lx = tl_b.ptx as integer;
    tl_ly = tl_b.pty as integer;
    bresenham(cv, tl_a.ptx as integer, tl_a.pty as integer, tl_lx, tl_ly, ink);
  }
  cv.set_pixel(tl_lx, tl_ly, ink);
}

/// The Bresenham walk itself, end-exclusive.
fn bresenham(cv: graphics::Canvas, bx0: integer, by0: integer, bx1: integer, by1: integer,
             ink: integer) {
  br_x = bx0;
  br_y = by0;
  br_dx = bx1 - bx0;
  br_xs = 1;
  if br_dx < 0 { br_dx = -br_dx; br_xs = -1; }
  br_dy = by1 - by0;
  br_ys = 1;
  if br_dy < 0 { br_dy = -br_dy; br_ys = -1; }
  if br_dx == 0 {
    for _ in 0..br_dy {
      cv.set_pixel(br_x, br_y, ink);
      br_y += br_ys;
    }
    return
  }
  if br_dy == 0 {
    for _ in 0..br_dx {
      cv.set_pixel(br_x, br_y, ink);
      br_x += br_xs;
    }
    return
  }
  if br_dx > br_dy {
    br_n = br_dx;
    br_dy += br_dy;
    br_e = br_dy - br_dx;
    br_dx += br_dx;
    for _ in 0..br_n {
      cv.set_pixel(br_x, br_y, ink);
      if br_e >= 0 { br_y += br_ys; br_e -= br_dx; }
      br_e += br_dy;
      br_x += br_xs;
    }
    return
  }
  br_n2 = br_dy;
  br_dx += br_dx;
  br_e2 = br_dx - br_dy;
  br_dy += br_dy;
  for _ in 0..br_n2 {
    cv.set_pixel(br_x, br_y, ink);
    if br_e2 >= 0 { br_x += br_xs; br_e2 -= br_dy; }
    br_e2 += br_dx;
    br_y += br_ys;
  }
}

// Copyright (c) 2026 Jurjen Stellingwerff
// SPDX-License-Identifier: LGPL-3.0-or-later
//
// scan — the byte-level reader the `.draw` line grammar is built out of.
//
// `draw.py` states that grammar as a dozen regular expressions, all of them
// `search` (match anywhere on the line) and all case-insensitive.  What matters
// for a port is not that this uses the same engine — it is that it accepts the
// same lines and reads the same numbers out of them, so the helpers here are
// shaped after the regex constructs rather than after a tidier grammar:
// `\b` is `word_boundary`, `[-\d.]+` is `read_number`, `(\d+)` is `read_uint`,
// and `\s*` is `skip_space`.
//
// Everything is BYTE-indexed.  A `.draw` command line is ASCII by construction
// (a line that is not is a comment, and comments never reach here), so a byte
// index and a character index are the same number — and `byte_at` reads without
// decoding, which is what makes a per-character scan of 300 lines cheap.

/// A parse position and what was read there.
///
/// A scan either lands (`ok`) or does not, and the caller needs BOTH the value
/// and where to carry on from, so every reader answers all three at once.
pub struct Scan {
  ok: boolean,
  at: integer,
  value: float,
}

pub fn scan_fail() -> Scan { Scan { ok: false, at: 0, value: 0.0 } }

pub fn is_digit(b: integer) -> boolean { b >= 48 && b <= 57 }

/// A regex `\w`: letters, digits and underscore.
pub fn is_word_byte(b: integer) -> boolean {
  (b >= 48 && b <= 57) || (b >= 65 && b <= 90) || (b >= 97 && b <= 122) || b == 95
}

/// A regex `\s`: space, tab, and the line endings a stray `\r` leaves behind.
pub fn is_space_byte(b: integer) -> boolean {
  b == 32 || b == 9 || b == 10 || b == 13 || b == 12 || b == 11
}

/// The byte at `i`, or -1 past either end — a value no byte can take, so a
/// caller never has to bound-check before asking.
pub fn at(s: text, i: integer) -> integer {
  if i < 0 || i >= size(s) { -1 } else { s.byte_at(i) }
}

/// ASCII lowercase of one byte.
pub fn lower_byte(b: integer) -> integer { if b >= 65 && b <= 90 { b + 32 } else { b } }

/// Past any run of spaces starting at `i`.
pub fn skip_space(s: text, i: integer) -> integer {
  sk_i = i;
  while is_space_byte(at(s, sk_i)) { sk_i += 1; }
  sk_i
}

/// Does `s` carry `word` at byte `i`, ignoring case?  (`word` must be lowercase.)
pub fn matches_at(s: text, i: integer, word: text) -> boolean {
  ma_n = size(word);
  if i + ma_n > size(s) { return false }
  for k in 0..ma_n {
    if lower_byte(at(s, i + k)) != word.byte_at(k) { return false }
  }
  true
}

/// A regex `\b` before byte `i`: position 0, or a non-word byte to the left.
pub fn word_boundary(s: text, i: integer) -> boolean {
  if i <= 0 { return true }
  !is_word_byte(at(s, i - 1))
}

/// A regex `([-\d.]+)` — a run of digits, dots and minus signs, read as a number.
///
/// The character class is what `draw.py` writes, so this is deliberately loose:
/// the run is taken first and converted second, and a run that is not a number
/// fails the scan rather than being silently read as zero.
pub fn read_number(s: text, i: integer) -> Scan {
  rn_j = i;
  while true {
    rn_b = at(s, rn_j);
    if is_digit(rn_b) || rn_b == 45 || rn_b == 46 { rn_j += 1; } else { break }
  }
  // The class swallows a `-` that is really the next token's separator, and the
  // regex engine backtracks off it because the pattern after demands it.  Give
  // back trailing bytes until what is left is a number, which lands on the same
  // answer without a second pass over the grammar.
  while rn_j > i {
    rn_v = s[i..rn_j] as float?;
    if rn_v != null { return Scan { ok: true, at: rn_j, value: rn_v ?? 0.0 } }
    rn_j -= 1;
  }
  scan_fail()
}

/// A regex `([\d.]+)` — digits and dots, no sign.
pub fn read_unsigned(s: text, i: integer) -> Scan {
  ru_j = i;
  while true {
    ru_b = at(s, ru_j);
    if is_digit(ru_b) || ru_b == 46 { ru_j += 1; } else { break }
  }
  while ru_j > i {
    ru_v = s[i..ru_j] as float?;
    if ru_v != null { return Scan { ok: true, at: ru_j, value: ru_v ?? 0.0 } }
    ru_j -= 1;
  }
  scan_fail()
}

/// A regex `(\d+)` — digits only.
pub fn read_uint(s: text, i: integer) -> Scan {
  rd_j = i;
  while is_digit(at(s, rd_j)) { rd_j += 1; }
  if rd_j == i { return scan_fail() }
  rd_v = s[i..rd_j] as integer?;
  if rd_v == null { return scan_fail() }
  Scan { ok: true, at: rd_j, value: (rd_v ?? 0) as float }
}

/// The first `\b<key>\s*=\s*` on the line, answering where its VALUE starts.
///
/// This is the shape every option regex in `draw.py` opens with, and the
/// word-boundary is load-bearing rather than decorative: without it `w=` would
/// also be found inside `w2=`, which is a different option with a different
/// meaning.  `key` must be lowercase; the match is not.
pub fn find_option(s: text, key: text) -> Scan {
  fo_n = size(s);
  fo_i = 0;
  while fo_i < fo_n {
    if word_boundary(s, fo_i) && matches_at(s, fo_i, key) {
      fo_j = skip_space(s, fo_i + size(key));
      if at(s, fo_j) == 61 {
        return Scan { ok: true, at: skip_space(s, fo_j + 1), value: 0.0 }
      }
    }
    fo_i += 1;
  }
  scan_fail()
}

/// The first occurrence of a lowercase `word` at or after `start`, ignoring case
/// — a regex literal with no boundary condition, which is what `Circle` and
/// `Line` open with.
///
/// The `start` is what makes a failed parse behave like `re.search`: a command
/// whose first occurrence does not parse is retried from the NEXT one rather
/// than given up on.
pub fn find_word_from(s: text, word: text, start: integer) -> Scan {
  fw_n = size(s);
  fw_i = if start < 0 { 0 } else { start };
  while fw_i < fw_n {
    if matches_at(s, fw_i, word) { return Scan { ok: true, at: fw_i + size(word), value: 0.0 } }
    fw_i += 1;
  }
  scan_fail()
}
